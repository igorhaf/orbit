"""
PROMPT #234 - Statistical Pattern Clusterer
Groups files by structural similarity using Jaccard index on symbol term sets.
No external ML dependencies - pure Python with collections and math.
"""

import logging
from dataclasses import dataclass, field
from typing import Dict, List, Set, Tuple
from collections import Counter

logger = logging.getLogger(__name__)

# ---------------------------------------------------------------------------
# Thresholds
# ---------------------------------------------------------------------------

SIMILARITY_THRESHOLD = 0.45
MIN_CLUSTER_SIZE = 3
HIGH_COHESION_THRESHOLD = 0.7
CONFIDENCE_BOOST = 0.1


@dataclass
class PatternCluster:
    """A cluster of structurally similar files."""
    cluster_id: str
    file_paths: List[str]
    centroid_terms: List[str] = field(default_factory=list)
    shared_functions: List[str] = field(default_factory=list)
    shared_imports: List[str] = field(default_factory=list)
    cohesion_score: float = 0.0
    size: int = 0
    representative_file: str = ""


class PatternClusterer:
    """
    Groups files by structural similarity using term frequency analysis.
    No AI calls. No external ML libraries.
    """

    def __init__(self):
        pass

    def cluster_files(
        self,
        symbols_by_file: Dict[str, Dict],
        file_groups: Dict,
    ) -> List[PatternCluster]:
        """
        Main entry. Takes symbol maps and groups files into structural clusters.
        """
        if not symbols_by_file:
            return []

        # Build feature vectors
        feature_vectors = self._build_feature_vectors(symbols_by_file)
        if len(feature_vectors) < MIN_CLUSTER_SIZE:
            return []

        # Group by extension for efficiency
        ext_groups: Dict[str, List[str]] = {}
        for fpath, symbols in symbols_by_file.items():
            ext = symbols.get("language", "unknown")
            ext_groups.setdefault(ext, []).append(fpath)

        all_clusters: List[PatternCluster] = []

        for ext, files in ext_groups.items():
            if len(files) < MIN_CLUSTER_SIZE:
                continue

            ext_vectors = {f: feature_vectors[f] for f in files if f in feature_vectors}
            if len(ext_vectors) < MIN_CLUSTER_SIZE:
                continue

            # Compute similarity and cluster
            sim_matrix = self._compute_similarity_matrix(ext_vectors)
            raw_clusters = self._cluster_by_similarity(sim_matrix, list(ext_vectors.keys()))

            for i, cluster_files in enumerate(raw_clusters):
                if len(cluster_files) < MIN_CLUSTER_SIZE:
                    continue

                cluster = self._characterize_cluster(
                    cluster_files, symbols_by_file, ext_vectors,
                    cluster_id=f"{ext}_{i}"
                )
                all_clusters.append(cluster)

        logger.info(
            f"Clustering: {len(all_clusters)} clusters from "
            f"{len(symbols_by_file)} files"
        )
        return all_clusters

    def _build_feature_vectors(
        self, symbols_by_file: Dict[str, Dict]
    ) -> Dict[str, Set[str]]:
        """Convert symbol maps into term sets for similarity comparison."""
        vectors: Dict[str, Set[str]] = {}

        for fpath, symbols in symbols_by_file.items():
            terms: Set[str] = set()

            # Imports
            for imp in symbols.get("imports", []):
                terms.add(f"imp:{imp.split('.')[-1]}")

            # Classes
            for cls in symbols.get("classes", []):
                terms.add(f"cls:{cls}")

            # Functions
            for func in symbols.get("functions", []):
                name = func["name"] if isinstance(func, dict) else str(func)
                terms.add(f"fn:{name}")

            # Decorators
            for dec in symbols.get("decorators", []):
                terms.add(f"dec:{dec}")

            # Constants
            for const in symbols.get("constants", []):
                terms.add(f"const:{const}")

            if terms:
                vectors[fpath] = terms

        return vectors

    def _compute_similarity_matrix(
        self, feature_vectors: Dict[str, Set[str]]
    ) -> Dict[Tuple[str, str], float]:
        """Compute pairwise Jaccard similarity."""
        matrix: Dict[Tuple[str, str], float] = {}
        keys = list(feature_vectors.keys())

        for i in range(len(keys)):
            for j in range(i + 1, len(keys)):
                a = feature_vectors[keys[i]]
                b = feature_vectors[keys[j]]
                union = a | b
                if not union:
                    continue
                jaccard = len(a & b) / len(union)
                if jaccard >= SIMILARITY_THRESHOLD:
                    matrix[(keys[i], keys[j])] = jaccard

        return matrix

    def _cluster_by_similarity(
        self,
        similarity_matrix: Dict[Tuple[str, str], float],
        file_keys: List[str],
    ) -> List[List[str]]:
        """Union-Find agglomerative clustering."""
        parent: Dict[str, str] = {k: k for k in file_keys}

        def find(x: str) -> str:
            while parent[x] != x:
                parent[x] = parent[parent[x]]
                x = parent[x]
            return x

        def union(a: str, b: str) -> None:
            ra, rb = find(a), find(b)
            if ra != rb:
                parent[ra] = rb

        # Union similar files
        for (a, b), sim in similarity_matrix.items():
            union(a, b)

        # Collect clusters
        clusters: Dict[str, List[str]] = {}
        for k in file_keys:
            root = find(k)
            clusters.setdefault(root, []).append(k)

        return list(clusters.values())

    def _characterize_cluster(
        self,
        cluster_files: List[str],
        symbols_by_file: Dict[str, Dict],
        feature_vectors: Dict[str, Set[str]],
        cluster_id: str,
    ) -> PatternCluster:
        """Compute centroid, shared terms, cohesion for a cluster."""
        # Centroid: most common terms
        term_counts: Counter = Counter()
        for fpath in cluster_files:
            if fpath in feature_vectors:
                for term in feature_vectors[fpath]:
                    term_counts[term] += 1

        total = len(cluster_files)
        centroid_terms = [
            t for t, c in term_counts.most_common(20)
            if c / total >= 0.5
        ]

        # Shared functions and imports
        shared_fns = [
            t.replace("fn:", "") for t in centroid_terms if t.startswith("fn:")
        ]
        shared_imps = [
            t.replace("imp:", "") for t in centroid_terms if t.startswith("imp:")
        ]

        # Cohesion: average pairwise Jaccard within cluster
        cohesion = self._compute_cohesion(cluster_files, feature_vectors)

        # Representative: file closest to centroid
        centroid_set = set(centroid_terms)
        best_file = cluster_files[0]
        best_overlap = 0
        for fpath in cluster_files:
            if fpath in feature_vectors:
                overlap = len(feature_vectors[fpath] & centroid_set)
                if overlap > best_overlap:
                    best_overlap = overlap
                    best_file = fpath

        return PatternCluster(
            cluster_id=cluster_id,
            file_paths=cluster_files,
            centroid_terms=[t for t in centroid_terms[:15]],
            shared_functions=shared_fns[:10],
            shared_imports=shared_imps[:10],
            cohesion_score=round(cohesion, 3),
            size=len(cluster_files),
            representative_file=best_file,
        )

    @staticmethod
    def _compute_cohesion(
        files: List[str], feature_vectors: Dict[str, Set[str]]
    ) -> float:
        """Average pairwise Jaccard similarity within cluster."""
        pairs = 0
        total_sim = 0.0

        for i in range(len(files)):
            a = feature_vectors.get(files[i], set())
            if not a:
                continue
            for j in range(i + 1, len(files)):
                b = feature_vectors.get(files[j], set())
                if not b:
                    continue
                union = a | b
                if union:
                    total_sim += len(a & b) / len(union)
                    pairs += 1

        return total_sim / pairs if pairs > 0 else 0.0

    def merge_with_static_patterns(
        self,
        clusters: List[PatternCluster],
        static_patterns: List,
    ) -> List:
        """
        Enrich static patterns with clustering info.
        High-cohesion clusters boost confidence of matching patterns.
        """
        if not clusters:
            return static_patterns

        # Build cluster lookup: file -> cluster
        file_to_cluster: Dict[str, PatternCluster] = {}
        for cluster in clusters:
            for fpath in cluster.file_paths:
                file_to_cluster[fpath] = cluster

        # Build group_key -> best cluster cohesion
        group_cohesion: Dict[str, float] = {}
        for cluster in clusters:
            # Determine which group_key this cluster maps to
            # Use first file's path to guess the group
            for fpath in cluster.file_paths:
                # Match against static patterns by group_key
                for pattern in static_patterns:
                    if any(fpath in ev for ev in getattr(pattern, "evidence", [])):
                        key = pattern.group_key
                        if key not in group_cohesion or cluster.cohesion_score > group_cohesion[key]:
                            group_cohesion[key] = cluster.cohesion_score

        # Boost confidence for patterns with high-cohesion clusters
        for pattern in static_patterns:
            cohesion = group_cohesion.get(pattern.group_key, 0.0)
            if cohesion >= HIGH_COHESION_THRESHOLD:
                old_conf = pattern.confidence_score
                pattern.confidence_score = min(
                    pattern.confidence_score + CONFIDENCE_BOOST, 1.0
                )
                pattern.is_high_confidence = pattern.confidence_score >= 0.75
                if pattern.confidence_score != old_conf:
                    pattern.key_characteristics.append(
                        f"cluster_cohesion:{cohesion:.2f}"
                    )
                    logger.debug(
                        f"Boosted {pattern.group_key} confidence "
                        f"{old_conf:.3f} -> {pattern.confidence_score:.3f}"
                    )

        return static_patterns
