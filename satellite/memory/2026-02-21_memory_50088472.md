<?php

namespace App\Http\Controllers\Aulas;

use App\Http\Controllers\Controller;
use App\Models\Lesson;
use Illuminate\Http\Request;

class LiveController extends Controller
{
	public function index(Request $request, string $id)
	{
		$lesson = $this->findLesson($id);
		abort_unless($lesson, 404);
		$course = $lesson->module->course;

		$curso = [ 'id' => $course->id, 'titulo' => $course->title ];
		$live = [
			'id' => $lesson->id,
			'tema' => 'Sessão ao vivo',
			'inicio' => now()->addMinutes(5)->format('Y-m-d H:i:s'),
			'fim' => now()->addMinutes(65)->format('Y-m-d H:i:s'),
			'status' => 'aguardando',
			'replayUrl' => null,
			'materiais_pre' => [],
			'materiais_pos' => [],
			'regras' => [],
		];

		return view('aulas.live', compact('curso','live'));
	}

	private function findLesson(string $id): ?Lesson
	{
		if (ctype_digit($id)) return Lesson::with('module.course')->find($id);
		return Lesson::with('module.course')->where('slug', $id)->first();
	}
}
