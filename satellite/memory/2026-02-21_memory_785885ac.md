namespace App\Http\Controllers\Instrutor;

use App\Http\Controllers\Controller;
use App\Models\Course;
use Illuminate\Http\Request;

class CuponsController extends Controller
{
    public function show(Request $request, string $id)
    {
        $user = $request->user();
        if (!$user || $user->role !== 'instrutor') {
            abort(403);
        }

        $course = Course::where('id', $id)
            ->where('instructor_id', $user->id)
            ->firstOrFail();

        $curso = [
            'id' => $course->id,
            'titulo' => $course->title,
            'cover' => $course->cover_url,
        ];

        // Por enquanto, dados mock para cupons (pode ser expandido depois com tabela real)
        $cupons = [
            [
                'id' => 1,
                'codigo' => 'LANCAMENTO2025',
                'desconto' => '20%',
                'tipo' => 'percentual',
                'valor' => 20,
                'uso_maximo' => 100,
                'usado' => 45,
                'validade' => '2025-12-31',
                'status' => 'ativo',
                'inscricoes' => 45,
                'receita_gerada' => 450.00,
            ],
            [
                'id' => 2,
                'codigo' => 'ESTUDANTE50',
                'desconto' => '50%',
                'tipo' => 'percentual',
                'valor' => 50,
                'uso_maximo' => 50,
                'usado' => 23,
                'validade' => '2025-06-30',
                'status' => 'ativo',
                'inscricoes' => 23,
                'receita_gerada' => 230.00,
            ],
        ];

        $resumo = [
            'total' => count($cupons),
            'ativos' => collect($cupons)->where('status', 'ativo')->count(),
            'total_inscricoes' => collect($cupons)->sum('inscricoes'),
            'receita_total' => collect($cupons)->sum('receita_gerada'),
        ];

        return view('instrutor.cursos.cupons', compact('curso', 'cupons', 'resumo'));
    }
}
