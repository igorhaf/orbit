<?php

namespace App\Http\Controllers\Instrutor;

use App\Http\Controllers\Controller;
use App\Models\Lesson;
use Illuminate\Http\Request;

class AulasLiveController extends Controller
{
    public function show(Request $request, string $id)
    {
        $user = $request->user();
        if (!$user || $user->role !== 'instrutor') {
            abort(403);
        }

        $lesson = Lesson::with(['module.course'])
            ->whereHas('module.course', function ($query) use ($user) {
                $query->where('instructor_id', $user->id);
            })
            ->findOrFail($id);

        $licao = [
            'id' => $lesson->id,
            'titulo' => $lesson->title,
            'descricao' => $lesson->description,
            'duracao' => $lesson->duration_minutes,
        ];

        $curso = [
            'id' => $lesson->module->course->id,
            'titulo' => $lesson->module->course->title,
        ];

        // Por enquanto, dados mock para a live (pode ser expandido depois)
        $live = [
            'status' => 'agendada',
            'data_inicio' => now()->addHours(2)->format('d/m/Y H:i'),
            'duracao' => '60 min',
            'sala' => 'https://meet.google.com/abc-defg-hij',
            'participantes' => 25,
            'gravacao' => false,
            'chat_ativo' => true,
        ];

        return view('instrutor.aulas.live', compact('licao', 'curso', 'live'));
    }
}
