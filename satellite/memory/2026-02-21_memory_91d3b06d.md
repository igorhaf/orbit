<?php

namespace App\Http\Controllers\Aulas;

use App\Http\Controllers\Controller;
use App\Models\Lesson;
use Illuminate\Http\Request;

class PlayerController extends Controller
{
	public function show(Request $request, string $id)
	{
		$lesson = $this->findLesson($id);
		abort_unless($lesson, 404);

		$course = $lesson->module->course;

		$curso = [ 'id' => $course->id, 'titulo' => $course->title ];
		$licaoAtual = [
			'id' => $lesson->id,
			'titulo' => $lesson->title,
			'duracao' => $lesson->duration_minutes,
			'descricao' => $lesson->description,
			'video' => $lesson->video_url,
		];
		$materiais = $lesson->materials()->get()->map(fn($m)=> [ 'nome'=>$m->name, 'tamanho'=> $m->size_kb ? round($m->size_kb/1024,2).' MB' : null, 'url'=>$m->url ])->all();

		$modulos = $course->modules()->with('lessons')->get()->map(function ($mod) {
			return [
				'titulo' => $mod->title,
				'aulas' => $mod->lessons->map(fn($l)=> [ 'id'=>$l->id, 'titulo'=>$l->title, 'duracao'=>$l->duration_minutes, 'lock'=>false ])->all(),
			];
		})->all();

		return view('aulas.player', compact('curso','licaoAtual','materiais','modulos'));
	}

	private function findLesson(string $id): ?Lesson
	{
		if (ctype_digit($id)) return Lesson::with('module.course','materials')->find($id);
		return Lesson::with('module.course','materials')->where('slug', $id)->first();
	}
}
