<?php

namespace App\Http\Controllers\Ajuda;

use App\Http\Controllers\Controller;
use App\Models\Faq;
use App\Models\FaqFeedback;
use Illuminate\Http\JsonResponse;
use Illuminate\Http\Request;

class FeedbackController extends Controller
{
    public function store(Request $request, Faq $faq): JsonResponse
    {
        $request->validate([
            'is_helpful' => ['required', 'boolean'],
        ]);

        $ip = $request->ip();

        $existing = FaqFeedback::query()
            ->where('faq_id', $faq->id)
            ->where('ip_address', $ip)
            ->first();

        if ($existing) {
            return response()->json([
                'message' => 'Você já avaliou este artigo.',
                'already_voted' => true,
            ]);
        }

        FaqFeedback::create([
            'faq_id' => $faq->id,
            'user_id' => $request->user()?->id,
            'is_helpful' => $request->boolean('is_helpful'),
            'ip_address' => $ip,
        ]);

        if ($request->boolean('is_helpful')) {
            $faq->increment('helpful_count');
        } else {
            $faq->increment('not_helpful_count');
        }

        return response()->json([
            'message' => 'Obrigado pelo seu feedback!',
            'already_voted' => false,
        ]);
    }
}
