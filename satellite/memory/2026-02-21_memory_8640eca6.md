<?php

namespace App\Http\Controllers\Aulas;

use App\Http\Controllers\Controller;
use App\Models\Lesson;
use App\Models\LessonDiscussion;
use Illuminate\Http\Request;

class DiscussõesController extends Controller
{
	public function index(Request $request, string $id)
	{
		$lesson = $this->findLesson($id);
		abort_unless($lesson, 404);
		$course = $lesson->module->course;

		$curso = [ 'id' => $course->id, 'titulo' => $course->title ];
		$licao = [ 'id' => $lesson->id, 'titulo' => $lesson->title ];
		$topicos = LessonDiscussion::where('lesson_id', $lesson->id)->latest()->get()->map(function(LessonDiscussion $d){
			return [
				'id' => $d->id,
				'titulo' => $d->title,
				'descricao' => $d->body,
				'tags' => $d->tags ?? [],
				'votos' => $d->votes_count,
				'respostas' => [],
				'autor' => $d->user?->name ?? 'Usuário',
				'tempo' => optional($d->created_at)->diffForHumans(),
				'resolvido' => !is_null($d->resolved_at),
			];
		})->all();
		$todasTags = collect($topicos)->flatMap(fn($t)=> $t['tags'])->unique()->values()->all();

		return view('aulas.discussoes', compact('curso','licao','topicos','todasTags'));
	}

	private function findLesson(string $id): ?Lesson
	{
		if (ctype_digit($id)) return Lesson::with('module.course')->find($id);
		return Lesson::with('module.course')->where('slug', $id)->first();
	}
}
