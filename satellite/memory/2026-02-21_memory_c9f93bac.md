<?php

namespace App\Policies;

use App\Models\Course;
use App\Models\User;

class CoursePolicy
{
	public function viewAny(?User $user): bool
	{
		return true; // cursos publicados são públicos
	}

	public function view(?User $user, Course $course): bool
	{
		if ($course->status === 'publicado') {
			return true;
		}
		return $user?->id === $course->instructor_id || $user?->role === 'instrutor';
	}

	public function create(User $user): bool
	{
		return $user->role === 'instrutor';
	}

	public function update(User $user, Course $course): bool
	{
		return $user->role === 'instrutor' && $user->id === $course->instructor_id;
	}

	public function delete(User $user, Course $course): bool
	{
		return $user->role === 'instrutor' && $user->id === $course->instructor_id;
	}
}
