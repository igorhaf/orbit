<?php

namespace App\Http\Controllers\Instrutor;

use App\Http\Controllers\Controller;
use App\Models\UserNotificationSetting;
use Illuminate\Http\Request;

class PreferenciasNotificacoesController extends Controller
{
    public function index(Request $request)
    {
        $user = $request->user();
        if (!$user || $user->role !== 'instrutor') {
            abort(403);
        }

        $settings = UserNotificationSetting::where('user_id', $user->id)->first();
        
        if (!$settings) {
            $settings = new UserNotificationSetting([
                'user_id' => $user->id,
                'settings' => [
                    'novos_alunos' => true,
                    'avaliacoes_baixas' => true,
                    'comentarios_duvidas' => true,
                    'lives_agendadas' => true,
                    'relatorios_mensais' => true,
                ],
                'summary_frequency' => 'diario',
                'quiet_hours' => ['start' => '22:00', 'end' => '08:00'],
            ]);
        }

        $preferencias = [
            'notificacoes' => [
                'novos_alunos' => $settings->settings['novos_alunos'] ?? true,
                'avaliacoes_baixas' => $settings->settings['avaliacoes_baixas'] ?? true,
                'comentarios_duvidas' => $settings->settings['comentarios_duvidas'] ?? true,
                'lives_agendadas' => $settings->settings['lives_agendadas'] ?? true,
                'relatorios_mensais' => $settings->settings['relatorios_mensais'] ?? true,
            ],
            'frequencia' => $settings->summary_frequency ?? 'diario',
            'horario_quieto' => $settings->quiet_hours ?? ['start' => '22:00', 'end' => '08:00'],
        ];

        return view('instrutor.preferencias-notificacoes', compact('preferencias'));
    }
}
