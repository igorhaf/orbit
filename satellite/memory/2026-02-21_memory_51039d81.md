<?php

namespace App\Http\Controllers\Instrutor;

use App\Http\Controllers\Controller;
use App\Http\Requests\Courses\CourseStatusRequest;
use App\Http\Requests\Courses\CourseStoreRequest;
use App\Http\Requests\Courses\CourseMetaRequest;
use App\Http\Requests\Courses\CourseUpdateRequest;
use App\Models\Course;
use Illuminate\Http\RedirectResponse;
use Illuminate\Http\Request;

class CourseController extends Controller
{
	public function index(Request $request): RedirectResponse
	{
		return redirect()->route('instrutor.meusCursos');
	}

	public function store(CourseStoreRequest $request): RedirectResponse
	{
		$this->authorize('create', Course::class);

		$attributes = $request->validated();
		$attributes['instructor_id'] = $request->user()->id;

		$course = Course::create($attributes);

		return redirect()->route('instrutor.meusCursos')
			->with('status', 'Curso criado com sucesso.');
	}

	public function update(CourseUpdateRequest $request, Course $course): RedirectResponse
	{
		$this->authorize('update', $course);

		$course->update($request->validated());

		return redirect()->route('instrutor.meusCursos')
			->with('status', 'Curso atualizado com sucesso.');
	}

	public function destroy(Request $request, Course $course): RedirectResponse
	{
		$this->authorize('delete', $course);

		$course->delete();

		return redirect()->route('instrutor.meusCursos')
			->with('status', 'Curso removido com sucesso.');
	}

	public function status(CourseStatusRequest $request, Course $course): RedirectResponse
	{
		$data = $request->validated();
		$course->status = $data['status'];
		$course->published_at = $data['status'] === 'publicado' ? now() : null;
		$course->save();

		return redirect()->route('instrutor.meusCursos')
			->with('status', 'Status do curso atualizado.');
	}

	public function meta(CourseMetaRequest $request, Course $course): RedirectResponse
	{
		$course->update($request->validated());
		return redirect()->route('instrutor.meusCursos')
			->with('status', 'Metadados do curso atualizados.');
	}
}
