<?php

namespace App\Http\Controllers;

use App\Models\User;
use App\Models\Course;
use Illuminate\Http\Request;

class InstrutoresController extends Controller
{
    public function index()
    {
        $instrutores = User::where('role', 'instrutor')
            ->get()
            ->map(function (User $instrutor) {
                $publishedCoursesQuery = $instrutor->courses()->where('status', 'publicado');

                $totalCursos = (int) $publishedCoursesQuery->count();
                $totalAlunos = (int) $publishedCoursesQuery->sum('enrollments_count');
                $notaMedia = (float) ($publishedCoursesQuery->avg('rating_average') ?? 0);

                return [
                    'id' => $instrutor->id,
                    'nome' => $instrutor->name,
                    'avatar' => $instrutor->avatar_url ?: 'https://images.unsplash.com/photo-1472099645785-5658abf4ff4e?w=400&h=400&fit=crop&crop=face',
                    'bio' => $instrutor->bio ?: 'Instrutor especializado em desenvolvimento de software.',
                    'total_cursos' => $totalCursos,
                    'total_alunos' => $totalAlunos,
                    'nota_media' => round($notaMedia, 1),
                    'especialidades' => $this->getEspecialidades($instrutor),
                ];
            })
            ->all();

        return view('instrutores.index', compact('instrutores'));
    }

    public function show(string $id)
    {
        $instrutor = User::where('id', $id)
            ->where('role', 'instrutor')
            ->firstOrFail();

        $perfil = [
            'id' => $instrutor->id,
            'nome' => $instrutor->name,
            'avatar' => $instrutor->avatar_url ?: 'https://images.unsplash.com/photo-1472099645785-5658abf4ff4e?w=400&h=400&fit=crop&crop=face',
            'bio' => $instrutor->bio ?: 'Instrutor especializado em desenvolvimento de software.',
            'email' => $instrutor->email,
        ];

        $cursosCollection = $instrutor->courses()
            ->where('status', 'publicado')
            ->latest('id')
            ->get()
            ->map(function (Course $course) {
                return [
                    'id' => $course->id,
                    'titulo' => $course->title,
                    'descricao' => $course->description ?: 'Sem descrição disponível.',
                    'cover' => $course->cover_url,
                    'categoria' => $course->category ?: 'Programação',
                    'nivel' => $course->level ?: 'Intermediário',
                    'duracao' => $course->duration_minutes,
                    'inscritos' => (int) ($course->enrollments_count ?? 0),
                    'nota' => round((float) ($course->rating_average ?? 0), 1),
                    'preco' => $course->price_cents ? $course->price_cents / 100 : 0,
                    'gratuito' => ($course->price_cents ?? 0) === 0,
                ];
            });

        $resumo = [
            'total_cursos' => $cursosCollection->count(),
            'total_alunos' => $cursosCollection->sum('inscritos'),
            'nota_media' => $cursosCollection->count() > 0
                ? round($cursosCollection->avg('nota'), 1)
                : 0,
            'especialidades' => $this->getEspecialidades($instrutor),
        ];

        $cursos = $cursosCollection->values()->all();

        return view('instrutores.show', compact('perfil', 'cursos', 'resumo'));
    }

    private function getEspecialidades(User $instrutor): array
    {
        $categorias = $instrutor->courses()
            ->where('status', 'publicado')
            ->pluck('category')
            ->filter()
            ->unique()
            ->values()
            ->all();

        if (empty($categorias)) {
            return ['Desenvolvimento Web', 'Programação'];
        }

        return array_slice($categorias, 0, 3);
    }
}
