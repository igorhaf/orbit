<?php

namespace App\Http\Controllers\Aulas;

use App\Http\Controllers\Controller;
use App\Models\Lesson;
use Illuminate\Http\Request;

class MateriaisController extends Controller
{
	public function index(Request $request, string $id)
	{
		$lesson = $this->findLesson($id);
		abort_unless($lesson, 404);
		$course = $lesson->module->course;

		$curso = [ 'id' => $course->id, 'titulo' => $course->title ];
		$licao = [ 'id' => $lesson->id, 'titulo' => $lesson->title ];
		$materiais = $lesson->materials()->get()->map(function($m){
			return [
				'id' => $m->id,
				'nome' => $m->name,
				'tipo' => $m->type,
				'tamanho' => $m->size_kb ? round($m->size_kb/1024,2).' MB' : '—',
				'atualizado' => optional($m->updated_at)->toDateString(),
				'tags' => $m->tags ?? [],
				'url' => $m->url,
			];
		})->all();

		$leituras = [];

		return view('aulas.materiais', compact('curso','licao','materiais','leituras'));
	}

	private function findLesson(string $id): ?Lesson
	{
		if (ctype_digit($id)) return Lesson::with('module.course','materials')->find($id);
		return Lesson::with('module.course','materials')->where('slug', $id)->first();
	}
}
