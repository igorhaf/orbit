<?php

namespace App\Http\Controllers\Trilhas;

use App\Http\Controllers\Controller;
use App\Models\Course;
use Illuminate\Http\Request;
use Illuminate\Support\Str;

class IndexController extends Controller
{
	public function index(Request $request)
	{
		$courses = Course::query()->published()->get([
			'id','title','tags','duration_minutes','level','rating_average','cover_url','category'
		]);

		$tagToCourses = [];
		foreach ($courses as $course) {
			$tags = (array) ($course->tags ?? []);
			foreach ($tags as $tag) {
				$k = trim((string) $tag);
				if ($k==='') continue;
				$tagToCourses[$k] = $tagToCourses[$k] ?? [];
				$tagToCourses[$k][] = $course;
			}
		}

		// Ordenar tags por número de cursos, decrescente
		uksort($tagToCourses, function($a,$b) use ($tagToCourses){
			return count($tagToCourses[$b]) <=> count($tagToCourses[$a]);
		});

		$trilhas = [];
		foreach ($tagToCourses as $tag => $list) {
			$durMin = collect($list)->sum(fn($c)=> (int) ($c->duration_minutes ?? 0));
			$durHoras = (int) round($durMin / 60);
			$nivel = collect($list)->pluck('level')->filter()->countBy()->sortDesc()->keys()->first() ?? 'Intermediário';
			$nota = (float) round(collect($list)->avg(fn($c)=> (float) ($c->rating_average ?? 0)), 1);
			$capa = collect($list)->pluck('cover_url')->filter()->first();
			if (!$capa) { $capa = 'https://picsum.photos/seed/'.(500 + (crc32(Str::slug($tag))%500)).'/1200/675'; }

			$trilhas[] = [
				'id' => Str::slug($tag),
				'nome' => 'Trilha '.$tag,
				'resumo' => 'Cursos focados em '.$tag.'.',
				'cursos' => count($list),
				'duracao' => max(1, $durHoras),
				'nivel' => $nivel ?? 'Intermediário',
				'nota' => $nota,
				'capa' => $capa,
				'recomendado' => count($list) >= 3 || $durHoras >= 10,
			];
		}

		$trilhas = array_slice($trilhas, 0, 9);

		return view('trilhas.index', compact('trilhas'));
	}
}
