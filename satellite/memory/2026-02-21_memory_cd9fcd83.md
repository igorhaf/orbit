"""
Rate Limiter Service
PROMPT #152 - Rate Limiting por Modelo de IA

Sliding Window Rate Limiter using Redis Sorted Sets.
Controls API request rate per AI model to avoid quota exceeded errors.
"""

import time
import logging
from typing import Dict, Optional, Tuple
from uuid import UUID

logger = logging.getLogger(__name__)


class RateLimiterService:
    """
    Sliding Window Rate Limiter using Redis.

    Each AI model can have its own rate limit configuration:
    - rate_limit_requests: Max number of requests allowed in the window
    - rate_limit_window_seconds: Time window in seconds

    Example: rate_limit_requests=3, rate_limit_window_seconds=60
    means "maximum 3 requests per minute"

    Uses Redis Sorted Sets for precise sliding window tracking:
    - Key: rate_limit:{model_id}
    - Score: Unix timestamp of the request
    - Member: Unique request ID (timestamp with microseconds)
    """

    def __init__(self, redis_client):
        """
        Initialize the rate limiter.

        Args:
            redis_client: Redis client instance (can be sync or async)
        """
        self.redis = redis_client
        self.key_prefix = "rate_limit"
        logger.info("✅ RateLimiterService initialized")

    def _get_key(self, model_id: str) -> str:
        """Get Redis key for a model's rate limit tracking."""
        return f"{self.key_prefix}:{model_id}"

    def check_rate_limit(
        self,
        model_id: str,
        max_requests: Optional[int],
        window_seconds: Optional[int]
    ) -> Tuple[bool, float]:
        """
        Check if a request can proceed based on rate limit.

        Args:
            model_id: UUID of the AI model (as string)
            max_requests: Maximum requests allowed in the window (None = no limit)
            window_seconds: Size of the sliding window in seconds (None = no limit)

        Returns:
            Tuple of (can_proceed: bool, wait_time_seconds: float)
            - can_proceed: True if request can be made now
            - wait_time_seconds: Seconds to wait if can_proceed is False
        """
        # No rate limit configured - allow immediately
        if max_requests is None or window_seconds is None:
            return (True, 0.0)

        if max_requests <= 0 or window_seconds <= 0:
            return (True, 0.0)

        try:
            # PROMPT #159 - Check provider backoff first
            # If provider returned "retry in Xs", respect that regardless of local count
            backoff_remaining = self.get_provider_backoff(model_id)
            if backoff_remaining > 0:
                logger.info(
                    f"⏸️ Provider backoff active for {model_id}: wait {backoff_remaining:.1f}s"
                )
                return (False, backoff_remaining)
            now = time.time()
            window_start = now - window_seconds
            key = self._get_key(model_id)

            # Remove old entries outside the current window
            self.redis.zremrangebyscore(key, 0, window_start)

            # Count current requests in window
            count = self.redis.zcard(key)

            if count < max_requests:
                # Under limit - can proceed
                logger.debug(f"✅ Rate limit OK for {model_id}: {count}/{max_requests} in last {window_seconds}s")
                return (True, 0.0)

            # At or over limit - calculate wait time
            # Get the oldest request in the window
            oldest = self.redis.zrange(key, 0, 0, withscores=True)
            if oldest:
                oldest_timestamp = oldest[0][1]
                # Wait until oldest request falls out of window
                wait_time = (oldest_timestamp + window_seconds) - now
                wait_time = max(0.0, wait_time)
                logger.info(
                    f"⏳ Rate limit reached for {model_id}: {count}/{max_requests} "
                    f"in last {window_seconds}s, wait {wait_time:.1f}s"
                )
                return (False, wait_time)

            # Shouldn't happen, but fallback to waiting full window
            logger.warning(f"⚠️ Rate limit edge case for {model_id}, waiting full window")
            return (False, float(window_seconds))

        except Exception as e:
            # If Redis fails, allow the request (fail-open)
            logger.error(f"❌ Rate limiter error for {model_id}: {e}")
            return (True, 0.0)

    def record_request(
        self,
        model_id: str,
        window_seconds: Optional[int]
    ) -> bool:
        """
        Record a request for rate limiting tracking.

        Should be called AFTER a successful API request.

        Args:
            model_id: UUID of the AI model (as string)
            window_seconds: Window size (for TTL calculation)

        Returns:
            True if recorded successfully, False otherwise
        """
        if window_seconds is None:
            return True

        try:
            now = time.time()
            key = self._get_key(model_id)

            # Use timestamp with microseconds as unique member
            member = f"{now:.6f}"

            # Add to sorted set with timestamp as score
            self.redis.zadd(key, {member: now})

            # Set TTL to auto-cleanup (2x window to be safe)
            self.redis.expire(key, window_seconds * 2)

            logger.debug(f"📝 Recorded request for {model_id}")
            return True

        except Exception as e:
            logger.error(f"❌ Failed to record request for {model_id}: {e}")
            return False

    def get_usage(
        self,
        model_id: str,
        max_requests: Optional[int],
        window_seconds: Optional[int]
    ) -> Dict:
        """
        Get current usage statistics for a model.

        Args:
            model_id: UUID of the AI model (as string)
            max_requests: Maximum requests allowed
            window_seconds: Window size in seconds

        Returns:
            Dict with usage statistics
        """
        if max_requests is None or window_seconds is None:
            return {
                "configured": False,
                "message": "Nenhum limite de taxa configurado"
            }

        try:
            now = time.time()
            window_start = now - window_seconds
            key = self._get_key(model_id)

            # Clean up and count
            self.redis.zremrangebyscore(key, 0, window_start)
            current_count = self.redis.zcard(key)

            # Get oldest request timestamp
            oldest = self.redis.zrange(key, 0, 0, withscores=True)
            oldest_timestamp = oldest[0][1] if oldest else None

            # Calculate time until next slot
            if current_count >= max_requests and oldest_timestamp:
                next_slot = (oldest_timestamp + window_seconds) - now
            else:
                next_slot = 0

            return {
                "configured": True,
                "model_id": model_id,
                "max_requests": max_requests,
                "window_seconds": window_seconds,
                "current_count": current_count,
                "remaining": max(0, max_requests - current_count),
                "next_slot_seconds": max(0, next_slot),
                "utilization_percent": (current_count / max_requests) * 100 if max_requests > 0 else 0
            }

        except Exception as e:
            logger.error(f"❌ Failed to get usage for {model_id}: {e}")
            return {
                "configured": True,
                "error": str(e)
            }

    def set_provider_backoff(self, model_id: str, backoff_seconds: float) -> bool:
        """
        Set a provider-mandated backoff period.

        When a provider returns "retry in Xs" (e.g., quota exceeded),
        this method blocks the model for that duration regardless of
        the local rate limit count.

        Args:
            model_id: UUID of the AI model (as string)
            backoff_seconds: How long to block (from provider's retry-after)

        Returns:
            True if set successfully
        """
        try:
            key = f"{self.key_prefix}:backoff:{model_id}"
            backoff_until = time.time() + backoff_seconds
            self.redis.set(key, str(backoff_until), ex=int(backoff_seconds) + 10)
            logger.warning(
                f"⏸️ Provider backoff set for {model_id}: {backoff_seconds:.1f}s "
                f"(until {time.strftime('%H:%M:%S', time.localtime(backoff_until))})"
            )
            return True
        except Exception as e:
            logger.error(f"❌ Failed to set backoff for {model_id}: {e}")
            return False

    def get_provider_backoff(self, model_id: str) -> float:
        """
        Check if there's an active provider backoff.

        Returns:
            Seconds remaining in backoff, or 0 if no backoff active
        """
        try:
            key = f"{self.key_prefix}:backoff:{model_id}"
            backoff_until = self.redis.get(key)
            if backoff_until:
                remaining = float(backoff_until) - time.time()
                if remaining > 0:
                    return remaining
            return 0.0
        except Exception as e:
            logger.error(f"❌ Failed to get backoff for {model_id}: {e}")
            return 0.0

    def clear_limit(self, model_id: str) -> bool:
        """
        Clear rate limit tracking for a model.
        Useful for testing or manual reset.

        Args:
            model_id: UUID of the AI model (as string)

        Returns:
            True if cleared successfully
        """
        try:
            key = self._get_key(model_id)
            self.redis.delete(key)
            logger.info(f"🗑️ Cleared rate limit for {model_id}")
            return True
        except Exception as e:
            logger.error(f"❌ Failed to clear limit for {model_id}: {e}")
            return False
