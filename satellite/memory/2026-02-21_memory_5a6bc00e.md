<?php

namespace App\Http\Controllers\Trilhas;

use App\Http\Controllers\Controller;
use App\Models\Course;
use Illuminate\Http\Request;
use Illuminate\Support\Str;

class ShowController extends Controller
{
	public function show(Request $request, string $id)
	{
		$tag = Str::of($id)->replace('-', ' ')->title()->toString();
		$courses = Course::query()->published()->with('instructor')
			->whereJsonContains('tags', $tag)
			->get();

		$durMin = (int) $courses->sum('duration_minutes');
		$durHoras = max(1, (int) round($durMin/60));
		$nivel = $courses->pluck('level')->filter()->countBy()->sortDesc()->keys()->first() ?? 'Intermediário';
		$nota = (float) round((float) $courses->avg('rating_average'), 1);
		$skills = $courses->pluck('tags')->flatten()->unique()->values()->all();
		$pre = ['Conhecimentos básicos do tema', 'Git e terminal', 'Curiosidade'];
		$regras = ['Ordem recomendada, mas não obrigatória','Critério de conclusão: 80% das aulas concluídas'];
		$descricao = 'Trilha com cursos focados em '.$tag.'.';

		$cursos = $courses->map(fn($c)=> [
			'id' => $c->id,
			'titulo' => $c->title,
			'duracao' => (int) max(1, round(($c->duration_minutes ?? 0)/60)),
			'status' => 'not-started',
			'nota' => (float) ($c->rating_average ?? 0),
			'badge' => null,
		])->values()->all();

		$trilha = [
			'id' => $id,
			'titulo' => 'Trilha '.$tag,
			'descricao' => $descricao,
			'objetivos' => ['Dominar fundamentos','Aplicar práticas em projetos','Preparar-se para avançado'],
			'duracaoTotal' => $durHoras,
			'nivel' => $nivel,
			'preRequisitos' => $pre,
			'nota' => $nota,
			'inscrito' => false,
			'progresso' => 0,
			'skills' => $skills,
			'regras' => $regras,
			'cursos' => $cursos,
			'instrutores' => $courses->pluck('instructor')->filter()->unique('id')->take(3)->map(fn($i)=> [
				'nome' => $i?->name,
				'bio' => $i?->bio ?? 'Instrutor na plataforma.',
				'foto' => $i?->avatar_url ?? 'https://i.pravatar.cc/120?u='.urlencode($i?->email ?? 'x'),
			])->values()->all(),
			'relacionadas' => [],
		];

		return view('trilhas.show', compact('trilha'));
	}
}
