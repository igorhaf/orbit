<?php

namespace App\Http\Controllers\Aluno;

use App\Http\Controllers\Controller;
use App\Models\CourseReview;
use App\Models\Enrollment;
use Illuminate\Http\Request;

class DashboardController extends Controller
{
	public function index(Request $request)
	{
		$user = $request->user();
		$aluno = [
			'nome' => $user?->name ?? 'Aluno',
			'avatar' => $user?->avatar_url ?? 'https://i.pravatar.cc/160?u='.urlencode($user?->email ?? 'anon'),
		];

		$enrolls = Enrollment::with('course')->where('user_id', $user?->id)->latest('last_accessed_at')->get();
		$ultimo = $enrolls->first();
		$ultimoCurso = $ultimo ? [
			'id' => $ultimo->course?->id,
			'titulo' => $ultimo->course?->title,
			'thumb' => $ultimo->course?->cover_url,
			'progresso' => $ultimo->progress_percent,
			'duracaoRestante' => '—',
		] : null;

		$emAndamento = $enrolls->take(5)->map(fn($e)=> [
			'id' => $e->course?->id,
			'titulo' => $e->course?->title,
			'thumb' => $e->course?->cover_url,
			'progresso' => $e->progress_percent,
		])->all();

		$recentes = CourseReview::with('course')->where('user_id', $user?->id)->latest()->take(5)->get()->map(fn($r)=> [
			'quando' => optional($r->created_at)->diffForHumans(),
			'acao' => 'Avaliou o curso "'.$r->course?->title.'" ('.$r->rating.'★)'
		])->all();

		$certificados = [];
		$proximasAtividades = [];
		$recomendacoes = [];

		return view('aluno.dashboard', compact('aluno','ultimoCurso','emAndamento','proximasAtividades','recentes','certificados','recomendacoes'));
	}
}
