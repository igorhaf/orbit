    def _extract_qa_pairs(self, conversation_data: list) -> Dict:
        """
        Extract Q&A pairs from meta prompt conversation.

        # Message 5: A2

                    contextual_qa.append({
                        "question": question["content"],
                        "answer": answer["content"]
                    })

        qa["contextual"] = contextual_qa

        return qa
