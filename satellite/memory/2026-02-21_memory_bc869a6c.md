	<script>
		(function(){
			const stateLoading = document.getElementById('stateLoading');
			const stateError = document.getElementById('stateError');
			const stateSuccess = document.getElementById('stateSuccess');
			const content = document.getElementById('content');
			const statusBadge = document.getElementById('statusBadge');
			const statusDesc = document.getElementById('statusDesc');
			const ctaPrimary = document.getElementById('ctaPrimary');
			const ctaSecondary = document.getElementById('ctaSecondary');
			const ctaSecondaryWrap = document.getElementById('ctaSecondaryWrap');
			const nextSteps = document.getElementById('nextSteps');

			let current = '{{ $inscricao['estado'] }}'; // pendente | confirmado | erro

			function setState(name){
				[stateLoading, stateError, content].forEach(el=> el.classList.add('hidden'));
				if (name==='loading') stateLoading.classList.remove('hidden');
				if (name==='error') stateError.classList.remove('hidden');
				if (name==='content') content.classList.remove('hidden');
			}

			function announce(msg){ stateSuccess.textContent = msg + ' 🎉'; stateSuccess.classList.remove('hidden'); setTimeout(()=> stateSuccess.classList.add('hidden'), 2000); }

			function applyStatusUI(){
				const cls = current==='confirmado' ? 'bg-emerald-500/20 text-emerald-300' : current==='pendente' ? 'bg-amber-500/20 text-amber-200' : 'bg-rose-500/20 text-rose-300';
				statusBadge.className = 'inline-flex items-center rounded-full px-2 py-0.5 text-xs ' + cls;
				statusBadge.textContent = current.charAt(0).toUpperCase()+current.slice(1);
				if (current==='confirmado'){
					statusDesc.textContent = 'Sua inscrição foi confirmada. Você já pode acessar o curso.';
					ctaPrimary.textContent = 'Ir para o curso';
					ctaPrimary.onclick = ()=> window.location.href = `{{ route('cursos.show', ':id') }}`.replace(':id', '{{ $id }}');
					nextSteps.textContent = 'Aproveite seus estudos! Esta inscrição aparece em Minhas Inscrições.';
					ctaSecondaryWrap.classList.add('hidden');
				} else if (current==='erro'){
					statusDesc.textContent = 'Não foi possível confirmar. Verifique suas informações e tente novamente.';
					ctaPrimary.textContent = 'Voltar ao checkout';
					ctaPrimary.onclick = ()=> history.back();
					nextSteps.textContent = 'Se o problema persistir, tente um método alternativo (UI).';
					ctaSecondary.textContent = 'Tentar novamente';
					ctaSecondaryWrap.classList.remove('hidden');
				} else {
					statusDesc.textContent = 'Processando pagamento. Assim que confirmar, você poderá acessar.';
					ctaPrimary.textContent = 'Atualizar status';
					ctaPrimary.onclick = ()=> { setState('loading'); setTimeout(()=>{ current = Math.random()>0.3 ? 'confirmado' : 'erro'; setState('content'); applyStatusUI(); announce('Status atualizado'); }, 800); };
					nextSteps.textContent = 'Aguarde a confirmação. Atualize esta página em alguns instantes.';
					ctaSecondaryWrap.classList.remove('hidden');
					ctaSecondary.textContent = 'Tentar novamente';
				}
			}

			// Fluxo inicial
			setState('loading'); setTimeout(()=>{ setState('content'); applyStatusUI(); }, 600);
		})();
	</script>
