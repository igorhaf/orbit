<?php

namespace App\Http\Controllers\Instrutor;

use App\Http\Controllers\Controller;
use App\Models\Course;
use App\Models\CourseReview;
use Illuminate\Http\Request;

class AvaliacoesController extends Controller
{
    public function show(Request $request, string $id)
    {
        $user = $request->user();
        if (!$user || $user->role !== 'instrutor') {
            abort(403);
        }

        $course = Course::where('id', $id)
            ->where('instructor_id', $user->id)
            ->firstOrFail();

        $curso = [
            'id' => $course->id,
            'titulo' => $course->title,
            'cover' => $course->cover_url,
        ];

        $avaliacoes = CourseReview::where('course_id', $course->id)
            ->with('user')
            ->latest()
            ->get()
            ->map(function (CourseReview $review) {
                return [
                    'id' => $review->id,
                    'aluno' => [
                        'nome' => $review->user?->name ?? 'Usuário',
                        'avatar' => $review->user?->avatar_url,
                    ],
                    'nota' => $review->rating,
                    'comentario' => $review->comment,
                    'data' => $review->created_at->toDateString(),
                    'tempo' => $review->created_at->diffForHumans(),
                    'status' => $review->rating >= 4 ? 'positiva' : ($review->rating <= 2 ? 'negativa' : 'neutra'),
                ];
            })
            ->all();

        $resumo = [
            'total' => count($avaliacoes),
            'positivas' => collect($avaliacoes)->where('status', 'positiva')->count(),
            'neutras' => collect($avaliacoes)->where('status', 'neutra')->count(),
            'negativas' => collect($avaliacoes)->where('status', 'negativa')->count(),
            'media' => $course->rating_average ?? 0,
            'total_avaliacoes' => $course->rating_count ?? 0,
        ];

        // Breakdown das notas
        $breakdown = $course->rating_breakdown ?? [5 => 0, 4 => 0, 3 => 0, 2 => 0, 1 => 0];
        $percentages = [];
        foreach ([5, 4, 3, 2, 1] as $star) {
            $count = (int) ($breakdown[$star] ?? 0);
            $percentages[$star] = $resumo['total_avaliacoes'] > 0 ? 
                round(($count / $resumo['total_avaliacoes']) * 100) : 0;
        }

        return view('instrutor.cursos.avaliacoes', compact('curso', 'avaliacoes', 'resumo', 'percentages'));
    }
}
