<?php

namespace App\Http\Controllers\Instrutor;

use App\Http\Controllers\Controller;
use App\Models\Course;
use App\Models\CourseReview;
use App\Models\Enrollment;
use App\Models\LessonDiscussion;
use Illuminate\Http\Request;

class PainelController extends Controller
{
    public function index(Request $request)
    {
        $user = $request->user();
        if (!$user || $user->role !== 'instrutor') {
            abort(403);
        }

        // Dados do instrutor
        $instrutor = [
            'nome' => $user->name,
            'avatar' => $user->avatar_url ?: 'https://randomuser.me/api/portraits/women/65.jpg',
        ];

        // Cards-resumo com dados reais
        $courses = Course::where('instructor_id', $user->id)->get();
        $resumo = [
            'cursosAtivos' => $courses->where('status', 'publicado')->count(),
            'alunosMatriculados' => $courses->sum('enrollments_count'),
            'avaliacoesRecentes' => CourseReview::whereIn('course_id', $courses->pluck('id'))
                ->where('created_at', '>=', now()->subDays(7))
                ->count(),
            'mediaNota' => $courses->avg('rating_average') ?: 0,
        ];

        // Atividade recente
        $atividade = collect();
        
        // Comentários recentes
        $recentDiscussions = LessonDiscussion::whereIn('lesson_id', 
            \App\Models\Lesson::whereIn('module_id', $courses->pluck('id'))->pluck('id')
        )->with('lesson.module.course')->latest()->take(3)->get();
        
        foreach ($recentDiscussions as $discussion) {
            $atividade->push([
                'tipo' => 'comentario',
                'quando' => $discussion->created_at->diffForHumans(),
                'texto' => "Novo comentário em \"{$discussion->lesson->module->course->title}\"",
            ]);
        }

        // Dúvidas recentes
        $recentReviews = CourseReview::whereIn('course_id', $courses->pluck('id'))
            ->where('rating', '<=', 3)
            ->with('course')
            ->latest()
            ->take(2)
            ->get();
            
        foreach ($recentReviews as $review) {
            $atividade->push([
                'tipo' => 'duvida',
                'quando' => $review->created_at->diffForHumans(),
                'texto' => "Avaliação baixa em \"{$review->course->title}\"",
            ]);
        }

        // Inscrições recentes
        $recentEnrollments = Enrollment::whereIn('course_id', $courses->pluck('id'))
            ->with('course')
            ->latest()
            ->take(2)
            ->get();
            
        foreach ($recentEnrollments as $enrollment) {
            $atividade->push([
                'tipo' => 'inscricao',
                'quando' => $enrollment->created_at->diffForHumans(),
                'texto' => "Nova inscrição em \"{$enrollment->course->title}\"",
            ]);
        }

        // Cursos em destaque
        $cursos = $courses->take(4)->map(function (Course $course) {
            return [
                'id' => $course->id,
                'titulo' => $course->title,
                'thumb' => $course->cover_url ?: 'https://images.unsplash.com/photo-1517433456452-f9633a875f6f?q=80&w=1200&auto=format&fit=crop',
                'inscritos' => (int) ($course->enrollments_count ?? 0),
                'nota' => (float) ($course->rating_average ?? 0),
            ];
        })->all();

        // Próximas ações (mock por enquanto)
        $acoes = [
            ['id' => 'a1', 'texto' => 'Revisar lições pendentes', 'feito' => false],
            ['id' => 'a2', 'texto' => 'Responder dúvidas dos alunos', 'feito' => false],
            ['id' => 'a3', 'texto' => 'Atualizar materiais dos cursos', 'feito' => true],
        ];

        // Agenda de lives (mock por enquanto)
        $lives = [
            ['data' => '2025-08-20 20:00', 'titulo' => 'Live: Q&A Laravel', 'sala' => '#'],
            ['data' => '2025-08-27 19:00', 'titulo' => 'Live: Tailwind avançado', 'sala' => '#'],
        ];

        return view('instrutor.painel', compact(
            'instrutor',
            'resumo',
            'atividade',
            'cursos',
            'acoes',
            'lives'
        ));
    }
}
