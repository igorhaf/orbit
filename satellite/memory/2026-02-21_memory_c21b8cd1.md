		if ($request->user()) {
			$enrollment = Enrollment::query()
				->with('course')
				->where('user_id', $request->user()->id)
				->orderByDesc('last_accessed_at')
				->first();
			if ($enrollment && $enrollment->course) {
				$progressoAtual = [
					'curso' => [
						'id' => $enrollment->course->id,
						'titulo' => $enrollment->course->title,
						'thumbnail' => $enrollment->course->cover_url ?: 'https://images.unsplash.com/photo-1515879218367-8466d910aaa4?q=80&w=1200&auto=format&fit=crop',

						'tempo_restante' => '—',
						'trilha' => [ 'id' => 1, 'titulo' => 'Minha trilha' ],
					],
					'momento' => $enrollment->last_position,
				];
			}
		}

		return view('home', compact('progressoAtual'));
	}
}
