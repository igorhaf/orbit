<?php

namespace App\Http\Controllers\Aluno;

use App\Http\Controllers\Controller;
use App\Models\Enrollment;
use Illuminate\Http\Request;

class InscricoesController extends Controller
{
	public function index(Request $request)
	{
		$user = $request->user();
		$recomendados = []; // manter vazio até termos recomendação real
		$inscricoes = Enrollment::with('course')
			->where('user_id', $user->id)
			->latest('created_at')
			->get()
			->map(function(Enrollment $e){
				return [
					'id' => $e->course?->id,
					'titulo' => $e->course?->title,
					'capa' => $e->course?->cover_url,
					'status' => $e->status === 'paid' ? 'concluida' : ($e->status === 'cancelled' ? 'cancelada' : 'ativa'),
					'data' => optional($e->created_at)?->toDateString(),
					'progresso' => $e->progress_percent ?? 0,
				];
			})->all();

		$tabs = ['todas','ativas','concluidas','canceladas'];
		$labels = ['todas'=>'Todas','ativas'=>'Ativas','concluidas'=>'Concluídas','canceladas'=>'Canceladas'];

		return view('aluno.inscricoes', compact('inscricoes','recomendados','tabs','labels'));
	}
}
