namespace App\Http\Requests\Courses;

use App\Models\Course;
use Illuminate\Foundation\Http\FormRequest;

class CourseMetaRequest extends FormRequest
{
	public function authorize(): bool
	{
		$course = $this->route('course');
		return $this->user()?->can('update', $course instanceof Course ? $course : new Course) ?? false;
	}

	public function rules(): array
	{
		return [
			// Requisitos / Público / Certificado
			'requirements' => ['nullable', 'array'],
			'requirements.*' => ['string', 'min:1', 'max:200'],
			'audience' => ['nullable', 'array'],
			'audience.*' => ['string', 'min:1', 'max:200'],
			'has_certificate' => ['nullable', 'boolean'],

			// O que você vai aprender (dinâmico por curso)
			'learning_outcomes' => ['nullable', 'array'],
			'learning_outcomes.*' => ['string', 'min:1', 'max:200'],

			// Programa (sílabus) opcional
			'syllabus' => ['nullable', 'array'],
			'syllabus.*.titulo' => ['required_with:syllabus', 'string', 'max:200'],
			'syllabus.*.aulas' => ['nullable', 'array'],
			'syllabus.*.aulas.*.titulo' => ['required_with:syllabus.*.aulas', 'string', 'max:200'],
			'syllabus.*.aulas.*.duracao' => ['nullable', 'integer', 'min:0', 'max:10000'],

			// Metadados auxiliares
			'tags' => ['nullable', 'array'],
			'tags.*' => ['string', 'min:1', 'max:50'],
			'category' => ['nullable', 'string', 'max:80'],
			'level' => ['nullable', 'string', 'max:20'],
		];
	}
}
