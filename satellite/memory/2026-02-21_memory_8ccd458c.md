<?php

namespace App\Http\Controllers\Instrutor;

use App\Http\Controllers\Controller;
use App\Models\Course;
use App\Models\Enrollment;
use Illuminate\Http\Request;

class CertificadosEmissaoLoteController extends Controller
{
    public function index(Request $request)
    {
        $user = $request->user();
        if (!$user || $user->role !== 'instrutor') {
            abort(403);
        }

        $courses = Course::where('instructor_id', $user->id)
            ->where('status', 'publicado')
            ->get();

        $cursos = $courses->map(function (Course $course) {
            $enrollments = $course->enrollments()
                ->where('status', 'ativo')
                ->where('progress_percent', '>=', 80) // 80% de progresso para certificado
                ->get();

            $certificadosEmitidos = $course->certificates()->count();
            $elegiveis = $enrollments->count() - $certificadosEmitidos;

            return [
                'id' => $course->id,
                'titulo' => $course->title,
                'cover' => $course->cover_url,
                'total_inscritos' => $enrollments->count(),
                'certificados_emitidos' => $certificadosEmitidos,
                'elegiveis' => max(0, $elegiveis),
                'selecionado' => false,
            ];
        })->all();

        $resumo = [
            'total_cursos' => count($cursos),
            'total_elegiveis' => collect($cursos)->sum('elegiveis'),
            'total_emitidos' => collect($cursos)->sum('certificados_emitidos'),
        ];

        return view('instrutor.certificados.emissao-lote', compact('cursos', 'resumo'));
    }
}
