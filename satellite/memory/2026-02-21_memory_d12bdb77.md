<?php

namespace App\Http\Controllers\Instrutor;

use App\Http\Controllers\Controller;
use App\Models\Course;
use Illuminate\Http\Request;

class MeusCursosController extends Controller
{
	public function index(Request $request)
	{
		$user = $request->user();
		if (!$user) {
			return redirect()->route('login');
		}
		if ($user->role !== 'instrutor') {
			abort(403);
		}

		$courses = Course::query()
			->where('instructor_id', $user->id)
			->latest('id')
			->get(['id','title','language','status','cover_url','duration_minutes','published_at']);

		$cursos = $courses->map(function (Course $course) {
			return [
				'id' => $course->id,
				'titulo' => $course->title,
				'categoria' => 'Programação',
				'nivel' => 'Intermediário',
				'idioma' => $this->mapLanguage($course->language),
				'thumb' => $course->cover_url ?: 'https://images.unsplash.com/photo-1515879218367-8466d910aaa4?q=80&w=1200&auto=format&fit=crop',
				'visibilidade' => $course->status === 'publicado' ? 'Público' : 'Privado',
				'inscritos' => (int) ($course->enrollments_count ?? 0),
				'nota' => (float) ($course->rating_average ?? 0),
				'badges' => $course->published_at && $course->published_at->gt(now()->subDays(30)) ? ['novo'] : [],
			];
		})->all();

		return view('instrutor.meus-cursos', [
			'instrutor' => [ 'nome' => $user->name ],
			'cursos' => $cursos,
		]);
	}

	private function mapLanguage(?string $code): string
	{
		return match ($code) {
			'en' => 'Inglês',
			'es' => 'Espanhol',
			default => 'Português',
		};
	}
}
