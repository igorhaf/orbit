<?php

namespace App\Http\Controllers\Cursos;

use App\Http\Controllers\Controller;
use App\Models\Course;
use Illuminate\Http\Request;

class ShowController extends Controller
{
	public function show(Request $request, string $id)
	{
		$course = $this->findCourse($id);
		abort_unless($course, 404);

		$this->authorize('view', $course);

		$curso = [
			'id' => $course->id,
			'titulo' => $course->title,
			'tags' => $course->tags ?: ['Programação', 'Backend', 'Laravel'],
			'rating' => (float) ($course->rating_average ?? 0),
			'alunos' => (int) ($course->enrollments_count ?? 0),
			'idioma' => $this->mapLanguage($course->language),
			'duracao' => (int) $course->duration_minutes,
			'preco' => $course->price_cents ? $course->price_cents / 100 : 0,
			'gratuito' => ($course->price_cents ?? 0) === 0,
			'descricao' => $course->description ?: 'Sem descrição informada.',
		];

		// O que você vai aprender — sempre vindo do banco (array), vazio se não definido
		$oQueAprendera = $course->learning_outcomes ?? [];

		$programa = $course->syllabus ?: [
			[ 'titulo' => 'Fundamentos e Setup', 'aulas' => [ ['titulo' => 'Introdução', 'duracao' => 8], ['titulo' => 'Setup', 'duracao' => 14] ] ],
			[ 'titulo' => 'Autenticação e Autorização', 'aulas' => [ ['titulo' => 'Login e sessões', 'duracao' => 18], ['titulo' => 'Policies e Gates', 'duracao' => 22] ] ],
		];

		$instrutores = [
			[
				'id' => $course->instructor_id,
				'nome' => $course->instructor?->name ?? 'Instrutor(a)',
				'foto' => $course->instructor?->avatar_url,
				'bio' => $course->instructor?->bio,
			],
		];

		$requisitos = $course->requirements ?: ['Noções de PHP', 'Composer instalado'];
		$publico = $course->audience ?: ['Devs iniciantes em Laravel'];
		$certificado = (bool) ($course->has_certificate ?? true);

		$breakdown = $course->rating_breakdown ?? [5=>0,4=>0,3=>0,2=>0,1=>0];
		$totalRatings = (int) ($course->rating_count ?? array_sum($breakdown));
		$percentages = [];
		foreach ([5,4,3,2,1] as $star) {
			$count = (int) ($breakdown[$star] ?? 0);
			$percentages[$star] = $totalRatings > 0 ? (int) round(($count / $totalRatings) * 100) : 0;
		}
		$avaliacoes = [
			'media' => (float) ($course->rating_average ?? 0),
			'total' => $totalRatings,
			'barras' => $percentages, // porcentagens p/ a UI
			'barras_count' => $breakdown, // counts brutos se necessário
		];

		$relacionados = [];

		return view('cursos.show', compact(
			'curso', 'oQueAprendera', 'programa', 'instrutores', 'requisitos', 'publico', 'certificado', 'avaliacoes', 'relacionados'
		));
	}

	private function findCourse(string $id): ?Course
	{
		if (ctype_digit($id)) {
			return Course::with('instructor')->find($id);
		}
		return Course::with('instructor')->where('slug', $id)->first();
	}

	private function mapLanguage(?string $code): string
	{
		return match ($code) {
			'en' => 'Inglês',
			'es' => 'Espanhol',
			default => 'Português',
		};
	}
}
