  source_file: string;
  rules_count: number;
}

interface RuleDetail {
  id: string;
  title: string;
  description: string;
  category: string;
  source: string;
  created_at: string;
}

  permission: 'bg-red-100 text-red-700',
  integration: 'bg-yellow-100 text-yellow-700',
  domain: 'bg-indigo-100 text-indigo-700',
  constraint: 'bg-orange-100 text-orange-700',
  general: 'bg-gray-100 text-gray-700',
};

export default function BusinessRulesPage() {
  const params = useParams();
  const router = useRouter();
  const projectId = params.id as string;

  const [project, setProject] = useState<any>(null);
  const [fileEntries, setFileEntries] = useState<RuleFileEntry[]>([]);
  const [totalRules, setTotalRules] = useState(0);
  const [loading, setLoading] = useState(true);
  const [activeFile, setActiveFile] = useState<string | null>(null);

  // Lazy-loaded rules per file
  const [rulesMap, setRulesMap] = useState<Map<string, RuleDetail[]>>(new Map());
  const [loadingRules, setLoadingRules] = useState<Set<string>>(new Set());
  const [expandedFiles, setExpandedFiles] = useState<Set<string>>(new Set());

  const sectionRefs = useRef<Record<string, HTMLDivElement | null>>({});

  const loadData = useCallback(async () => {
    setLoading(true);
    try {
      const [proj, rulesByFile] = await Promise.all([
        projectsApi.get(projectId),
        knowledgeApi.getRulesByFile(projectId),
      ]);
      setProject(proj.data || proj);
      setFileEntries(rulesByFile.files);
      setTotalRules(rulesByFile.total_rules);
    } catch (error) {
      console.error('Failed to load rules:', error);
    } finally {
      setLoading(false);
    }
  }, [projectId]);

  useEffect(() => {
    loadData();
  }, [loadData]);

  const scrollToFile = (sourceFile: string) => {
    setActiveFile(sourceFile);
    const el = sectionRefs.current[sourceFile];
    if (el) {
      el.scrollIntoView({ behavior: 'smooth', block: 'start' });
    }
  };

  const toggleFileRules = async (sourceFile: string) => {
    if (expandedFiles.has(sourceFile)) {
      setExpandedFiles((prev) => {
        const n = new Set(prev);
        n.delete(sourceFile);
        return n;
      });
      return;
    }

    // Lazy load rules for this file
    if (!rulesMap.has(sourceFile)) {
      setLoadingRules((prev) => new Set(prev).add(sourceFile));
      try {
        const rules = await knowledgeApi.listRules(projectId, {
          source_file: sourceFile === '(sem arquivo)' ? undefined : sourceFile,
          limit: 500,
        });
        setRulesMap((prev) => new Map(prev).set(sourceFile, rules));
      } catch (error) {
        console.error('Failed to load rules for file:', error);
        setRulesMap((prev) => new Map(prev).set(sourceFile, []));
      } finally {
        setLoadingRules((prev) => {
          const n = new Set(prev);
          n.delete(sourceFile);
          return n;
        });
      }
    }
    setExpandedFiles((prev) => new Set(prev).add(sourceFile));
  };

  const getShortPath = (fullPath: string) => {
    if (fullPath === '(sem arquivo)') return fullPath;
    // Show last 2-3 segments
    const parts = fullPath.split('/');
    return parts.length > 3 ? '.../' + parts.slice(-3).join('/') : fullPath;
  };

  if (loading) {

      <Layout>
        <div className="flex items-center justify-center h-64">
          <div className="animate-spin rounded-full h-12 w-12 border-b-2 border-blue-600"></div>
        </div>
      </Layout>
    );
  }

            <div className="flex items-center justify-between mb-3">
              <h2 className="text-xs font-semibold text-gray-500 uppercase tracking-wider">Arquivos</h2>
              <span className="text-xs text-gray-400">{totalRules} regras</span>
            </div>
            <div className="space-y-0.5 max-h-[70vh] overflow-y-auto">
              {fileEntries.map((entry) => (
                <button
                  key={entry.source_file}
                  onClick={() => scrollToFile(entry.source_file)}
                  className={`w-full text-left px-3 py-1.5 rounded text-sm transition-colors flex items-center justify-between gap-1 ${
                    activeFile === entry.source_file
                      ? 'bg-orange-100 text-orange-700 font-medium'
                      : 'text-gray-600 hover:bg-gray-100 hover:text-gray-900'
                  }`}
                  title={entry.source_file}
                >

                  <span className={`text-xs px-1.5 py-0.5 rounded-full flex-shrink-0 ${
                    activeFile === entry.source_file ? 'bg-orange-200 text-orange-800' : 'bg-gray-200 text-gray-600'
                  }`}>
                    {entry.rules_count}
                  </span>
                </button>
              ))}
            </div>
          </Card>
        </div>

        {/* Main Content */}
        <div className="flex-1 min-w-0">
          {/* Page Header */}
          <div className="flex items-center justify-between mb-6">
            <div className="flex items-center gap-3">
              <button
                onClick={() => router.push(`/projects/${projectId}`)}
                className="text-gray-400 hover:text-gray-600"
              >
                <svg className="w-5 h-5" fill="none" stroke="currentColor" viewBox="0 0 24 24">
                  <path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M15 19l-7-7 7-7" />

              <span className="px-2 py-0.5 rounded-full text-xs bg-orange-100 text-orange-700">{totalRules} regras</span>
