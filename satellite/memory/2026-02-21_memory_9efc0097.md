namespace App\Http\Controllers\Instrutor;

use App\Http\Controllers\Controller;
use App\Models\Course;
use App\Models\Enrollment;
use App\Models\Lesson;
use Illuminate\Http\Request;

class EstatisticasController extends Controller
{
    public function show(Request $request, string $id)
    {
        $user = $request->user();
        if (!$user || $user->role !== 'instrutor') {
            abort(403);
        }

        $course = Course::where('id', $id)
            ->where('instructor_id', $user->id)
            ->with(['modules.lessons', 'enrollments'])
            ->firstOrFail();

        $curso = [
            'id' => $course->id,
            'titulo' => $course->title,
            'cover' => $course->cover_url,
        ];

        // KPIs principais
        $kpis = [
            'inscritos' => $course->enrollments()->where('status', 'ativo')->count(),
            'conclusao_media' => $this->calcularConclusaoMedia($course),
            'nota_media' => (float) ($course->rating_average ?? 0),
            'tempo_estudo_medio' => $this->calcularTempoEstudoMedio($course),
        ];

        // Progresso por módulo
        $progressoModulos = $course->modules->map(function ($module) use ($course) {
            $totalLicoes = $module->lessons->count();
            $licoesCompletadas = $this->calcularLicoesCompletadas($module, $course);
            
            return [
                'titulo' => $module->title,
                'total_licoes' => $totalLicoes,
                'completadas' => $licoesCompletadas,
                'percentual' => $totalLicoes > 0 ? round(($licoesCompletadas / $totalLicoes) * 100) : 0,
            ];
        })->all();

        // Lições mais vistas
        $licoesMaisVistas = $course->modules->flatMap(function ($module) {
            return $module->lessons;
        })->map(function (Lesson $lesson) {
            return [
                'id' => $lesson->id,
                'titulo' => $lesson->title,
                'modulo' => $lesson->module->title,
                'visualizacoes' => rand(50, 200), // Mock por enquanto
                'abandono' => rand(5, 25), // Mock por enquanto
            ];
        })->sortByDesc('visualizacoes')
          ->take(5)
          ->values()
          ->all();

        // Evolução de inscritos (últimos 30 dias)
        $evolucaoInscritos = $this->gerarEvolucaoInscritos($course);

        return view('instrutor.cursos.estatisticas', compact(
            'curso',
            'kpis',
            'progressoModulos',
            'licoesMaisVistas',
            'evolucaoInscritos'
        ));
    }

    private function calcularConclusaoMedia(Course $course): float
    {
        $enrollments = $course->enrollments()->where('status', 'ativo')->get();
        if ($enrollments->isEmpty()) return 0;

        $totalProgresso = $enrollments->sum('progress_percent');
        return round($totalProgresso / $enrollments->count(), 1);
    }

    private function calcularTempoEstudoMedio(Course $course): string
    {
        $enrollments = $course->enrollments()->where('status', 'ativo')->get();
        if ($enrollments->isEmpty()) return '0 min';

        $totalMinutos = $enrollments->sum(function ($enrollment) {
            return $enrollment->last_position ?? 0;
        });

        $mediaMinutos = $totalMinutos / $enrollments->count();
        
        if ($mediaMinutos >= 60) {
            $horas = floor($mediaMinutos / 60);
            $minutos = round($mediaMinutos % 60);
            return "{$horas}h {$minutos}min";
        }
        
        return round($mediaMinutos) . ' min';
    }

    private function calcularLicoesCompletadas($module, Course $course): int
    {
        // Mock por enquanto - pode ser expandido com tracking real
        return rand(0, $module->lessons->count());
    }

    private function gerarEvolucaoInscritos(Course $course): array
    {
        $evolucao = [];
        $data = now()->subDays(30);
        
        for ($i = 0; $i < 30; $i++) {
            $evolucao[] = [
                'data' => $data->copy()->addDays($i)->format('d/m'),
                'inscritos' => rand(10, 50), // Mock por enquanto
            ];
        }
        
        return $evolucao;
    }
}
