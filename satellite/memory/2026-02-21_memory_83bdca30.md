TAG_MAP: List[tuple] = [

# File path detection pattern
FILE_PATH_RE = re.compile(
    r"(?:^|\s|[`\"'])"
    r"((?:[\w\-]+/)+[\w\-]+\.[\w]+)"
    r"(?:\s|$|[`\"':,])",
    re.MULTILINE,
)

# Tier configuration
TIER_CONFIG = {
    "trivial": {"tier": "fast", "tokens": 150},
    "simple": {"tier": "fast", "tokens": 200},
    "moderate": {"tier": "balanced", "tokens": 350},
    "complex": {"tier": "strong", "tokens": 500},
}


def analyze_commit_complexity(
    changes_text: str,
    task_title: Optional[str] = None,
    task_description: Optional[str] = None,
) -> Dict:
    """
    Analyze change complexity to determine model routing tier.

    Args:
        changes_text: The summarized changes text
        task_title: Optional task title for additional context
        task_description: Optional task description

    Returns:
        Dict with complexity, estimated_tokens, recommended_tier,
        file_count, change_density, semantic_tags, is_refactor, is_multi_module
    """
    combined = changes_text or ""
    if task_title:
        combined = f"{task_title}\n{combined}"
    if task_description:
        combined = f"{combined}\n{task_description}"

    combined_lower = combined.lower()
    words = set(re.findall(r"\b[a-z_]{2,}\b", combined_lower))

    # --- File count ---
    file_matches = FILE_PATH_RE.findall(combined)
    file_count = len(set(file_matches))

    # --- Change density (0.0 - 1.0) ---
    text_len = len(changes_text or "")
    change_density = min(text_len / 800.0, 1.0)

    # --- Keyword detection ---
    refactor_hits = words & REFACTOR_KEYWORDS
    multi_module_hits = words & MULTI_MODULE_KEYWORDS
    security_hits = words & SECURITY_KEYWORDS
    perf_hits = words & PERFORMANCE_KEYWORDS

    is_refactor = len(refactor_hits) >= 1
    is_multi_module = len(multi_module_hits) >= 2
    technical_keyword_count = (
        len(refactor_hits) + len(multi_module_hits)
        + len(security_hits) + len(perf_hits)
    )

    # --- Semantic tags ---
    semantic_tags: List[str] = []
    for kw_set, tag in TAG_MAP:
        if words & kw_set:
            semantic_tags.append(tag)

    # --- Complexity classification ---
    if text_len < 100 and file_count <= 1 and technical_keyword_count == 0:
        complexity = "trivial"
    elif text_len < 300 and file_count <= 2 and not is_refactor:
        complexity = "simple"
    elif (
        text_len > 600
        or file_count > 5
        or (is_refactor and is_multi_module)
        or technical_keyword_count >= 6
    ):
        complexity = "complex"
    else:
        complexity = "moderate"

    tier_info = TIER_CONFIG[complexity]

    result = {
        "complexity": complexity,
        "estimated_tokens": tier_info["tokens"],
        "recommended_tier": tier_info["tier"],
        "file_count": file_count,
        "change_density": round(change_density, 2),
        "semantic_tags": semantic_tags,
        "is_refactor": is_refactor,
        "is_multi_module": is_multi_module,
    }

    logger.info(
        f"Commit complexity: {complexity} | tier={tier_info['tier']} | "
        f"files={file_count} density={change_density:.2f} "
        f"keywords={technical_keyword_count} tags={semantic_tags}"
    )

    return result
