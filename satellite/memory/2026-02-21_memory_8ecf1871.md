        (function(){
            const RAW = @json($interacoes);

            const tbody = document.getElementById('tbody');
            const empty = document.getElementById('stateEmpty');

            function badgeStatus(s){ return s==='resolvido' ? 'bg-emerald-500/20 text-emerald-300' : 'bg-white/10 text-white/70'; }
            function renderStats(votos, resp){
                return `<span class="sr-only">${votos} votos, ${resp} respostas</span><span aria-hidden="true" class="inline-flex items-center gap-2"><span>▲ ${votos}</span><span>💬 ${resp}</span></span>`;
            }

            function filter(){
                const tipo = filtro.value;
                const term = (busca.value||'').toLowerCase();
                return RAW.filter(i=> (tipo==='all' || i.tipo===tipo) && (!term || i.topico.toLowerCase().includes(term)));
            }

            function render(){
                const list = filter();
                tbody.innerHTML='';
                if (list.length===0){ empty.classList.remove('hidden'); return; }
                empty.classList.add('hidden');
                list.forEach(i=>{
                    const tr=document.createElement('tr'); tr.className='border-t border-white/10';
                    tr.innerHTML = `
                        <td class="py-3">
                            <div class="font-medium line-clamp-1">${i.topico}</div>
                            <div class="text-xs text-white/60">${i.tipo==='post'?'Minha postagem': i.tipo==='reply'?'Minha resposta':'Seguido'}</div>
                        </td>
                        <td class="py-3 text-white/70">${i.curso} • ${i.aula}</td>
                        <td class="py-3 text-white/70">${new Date(i.data).toLocaleDateString('pt-BR')}</td>
                        <td class="py-3"><span class="inline-flex items-center rounded-full px-2 py-0.5 text-xs ${badgeStatus(i.status)}">${i.status}</span></td>
                        <td class="py-3">${renderStats(i.votos, i.respostas)}</td>
                        <td class="py-3">
                            <div class="flex items-center gap-2 text-xs">
                                <a href="${`{{ route('aulas.discussoes', ':id') }}`.replace(':id','{{ $licao['id'] }}')}" class="rounded-md border border-white/15 px-2 py-1 hover:bg-white/5 focus:outline-none focus:ring-2 focus:ring-fuchsia-500/40" aria-label="Abrir thread ${i.topico}">Abrir thread</a>
                                <button class="btnFollow rounded-md border border-white/15 px-2 py-1 hover:bg-white/5 focus:outline-none focus:ring-2 focus:ring-fuchsia-500/40">${i.seguindo?'Deixar de seguir':'Seguir'}</button>
