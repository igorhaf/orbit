<?php

namespace App\Http\Controllers\Aluno;

use App\Http\Controllers\Controller;
use App\Models\Course;
use App\Models\CourseReview;
use Illuminate\Http\Request;

class MinhaNotaController extends Controller
{
	public function show(Request $request, string $id)
	{
		$user = $request->user();
		$course = Course::with('instructor')->findOrFail($id);
		$curso = [
			'id' => $course->id,
			'titulo' => $course->title,
			'capa' => $course->cover_url,
		];
		$review = CourseReview::where('course_id',$course->id)->where('user_id',$user->id)->latest()->first();
		if (!$review) {
			$estado = 'empty';
			return view('aluno.minha-nota', compact('curso','estado')); 
		}
		$estado = 'ok';
		$minha = [
			'nota' => (int) $review->rating,
			'data' => optional($review->published_at ?? $review->created_at)?->toDateString(),
			'titulo' => $review->title,
		];
		$historico = CourseReview::where('course_id',$course->id)->where('user_id',$user->id)->orderBy('created_at')->get()->map(fn($r)=> [
			'data' => optional($r->created_at)?->toDateString(),
			'nota' => (int) $r->rating,
		])->all();
		return view('aluno.minha-nota', compact('curso','estado','minha','historico'));
	}
}
