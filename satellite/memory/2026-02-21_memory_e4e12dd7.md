<?php

namespace App\Http\Controllers\Aluno;

use App\Http\Controllers\Controller;
use App\Models\Enrollment;
use Illuminate\Http\Request;

class MeusCursosController extends Controller
{
	public function index(Request $request)
	{
		$user = $request->user();
		$aluno = [ 'nome' => $user->name ];
		$enrolls = Enrollment::with(['course.instructor'])->where('user_id',$user->id)->get();
		$cursos = $enrolls->map(function(Enrollment $e){
			$course = $e->course; if(!$course) return null;
			return [
				'id' => $course->id,
				'titulo' => $course->title,
				'categoria' => $course->category,
				'instrutor' => $course->instructor?->name,
				'nivel' => $course->level,
				'idioma' => $course->language === 'pt' ? 'Português' : strtoupper($course->language),
				'preco' => ($course->price_cents ?? 0) / 100,
				'thumb' => $course->cover_url,
				'status' => ($e->progress_percent ?? 0) >= 100 ? 'completed' : 'in_progress',
				'progresso' => $e->progress_percent ?? 0,
				'certificado' => false,
				'favorito' => false,
			];
		})->filter()->values()->all();

		$categorias = collect($cursos)->pluck('categoria')->filter()->unique()->values()->all();
		$niveis = collect($cursos)->pluck('nivel')->filter()->unique()->values()->all();
		$idiomas = collect($cursos)->pluck('idioma')->filter()->unique()->values()->all();

		return view('aluno.meus-cursos', compact('aluno','cursos','categorias','niveis','idiomas'));
	}
}
