def get_db() -> Generator[Session, None, None]:
    """
    Dependency function to get database session

        @app.get("/items")
        def read_items(db: Session = Depends(get_db)):
            return db.query(Item).all()
    """
    db = SessionLocal()
    try:
        yield db
    finally:
        db.close()
