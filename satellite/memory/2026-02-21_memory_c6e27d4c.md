			function avatar(name){ return `https://i.pravatar.cc/40?u=${encodeURIComponent(name)}`; }

			function sortData(list){ const o=ordSel.value; if(o==='high') return list.slice().sort((a,b)=> b.nota-a.nota); if(o==='low') return list.slice().sort((a,b)=> a.nota-b.nota); return list.slice().sort((a,b)=> new Date(b.data)-new Date(a.data)); }
