<?php

namespace App\Http\Controllers\Instrutor;

use App\Http\Controllers\Controller;
use App\Models\Lesson;
use App\Models\Module;
use App\Models\Course;
use Illuminate\Http\Request;

class LicoesEditarController extends Controller
{
    public function edit(Request $request, string $cursoId, string $moduloId, string $licaoId)
    {
        $user = $request->user();
        if (!$user || $user->role !== 'instrutor') {
            abort(403);
        }

        $course = Course::where('id', $cursoId)
            ->where('instructor_id', $user->id)
            ->firstOrFail();

        $module = Module::where('id', $moduloId)
            ->where('course_id', $course->id)
            ->firstOrFail();

        $lesson = Lesson::where('id', $licaoId)
            ->where('module_id', $module->id)
            ->with('materials')
            ->firstOrFail();

        $curso = [
            'id' => $course->id,
            'titulo' => $course->title,
        ];

        $modulo = [
            'id' => $module->id,
            'titulo' => $module->title,
        ];

        $licao = [
            'id' => $lesson->id,
            'titulo' => $lesson->title,
            'descricao' => $lesson->description,
            'tipo' => $lesson->type ?? 'video',
            'duracao' => $lesson->duration_minutes,
            'status' => $lesson->status ?? 'rascunho',
            'video_url' => $lesson->video_url,
            'materiais' => $lesson->materials->map(function ($material) {
                return [
                    'id' => $material->id,
                    'nome' => $material->name,
                    'tipo' => $material->type,
                    'url' => $material->url,
                    'tamanho' => $material->size_kb ? round($material->size_kb / 1024, 2) . ' MB' : '—',
                ];
            })->all(),
        ];

        return view('instrutor.licoes.editar', compact('curso', 'modulo', 'licao'));
    }
}
