/**
 * Business Rules Browser
 * PROMPT #238 - Individual page showing all business rules grouped by source file
 *
 * Wiki-style layout with sidebar listing source files and main content
 * showing rules grouped by file.
 */

'use client';

import { useEffect, useState, useCallback, useRef } from 'react';
import { useParams, useRouter } from 'next/navigation';
import { Layout, Breadcrumbs } from '@/components/layout';
import { Card } from '@/components/ui';
import { projectsApi, knowledgeApi } from '@/lib/api';

interface RuleFileEntry {
  source_file: string;
  rules_count: number;
}

interface RuleDetail {
  id: string;
  title: string;
  description: string;
  category: string;
  source: string;
  created_at: string;
}

const CATEGORY_COLORS: Record<string, string> = {
  validation: 'bg-blue-100 text-blue-700',
  workflow: 'bg-purple-100 text-purple-700',
  calculation: 'bg-green-100 text-green-700',
  permission: 'bg-red-100 text-red-700',
  integration: 'bg-yellow-100 text-yellow-700',
  domain: 'bg-indigo-100 text-indigo-700',
  constraint: 'bg-orange-100 text-orange-700',
  general: 'bg-gray-100 text-gray-700',
};

export default function BusinessRulesPage() {
  const params = useParams();
  const router = useRouter();
  const projectId = params.id as string;

  const [project, setProject] = useState<any>(null);
  const [fileEntries, setFileEntries] = useState<RuleFileEntry[]>([]);
  const [totalRules, setTotalRules] = useState(0);
  const [loading, setLoading] = useState(true);
  const [activeFile, setActiveFile] = useState<string | null>(null);

  // Lazy-loaded rules per file
  const [rulesMap, setRulesMap] = useState<Map<string, RuleDetail[]>>(new Map());
  const [loadingRules, setLoadingRules] = useState<Set<string>>(new Set());
  const [expandedFiles, setExpandedFiles] = useState<Set<string>>(new Set());

  const sectionRefs = useRef<Record<string, HTMLDivElement | null>>({});

  const loadData = useCallback(async () => {
    setLoading(true);
    try {
      const [proj, rulesByFile] = await Promise.all([
        projectsApi.get(projectId),
        knowledgeApi.getRulesByFile(projectId),
      ]);
      setProject(proj.data || proj);
      setFileEntries(rulesByFile.files);
      setTotalRules(rulesByFile.total_rules);
    } catch (error) {
      console.error('Failed to load rules:', error);
    } finally {
      setLoading(false);
    }
  }, [projectId]);

  useEffect(() => {
    loadData();
  }, [loadData]);

  const scrollToFile = (sourceFile: string) => {
    setActiveFile(sourceFile);
    const el = sectionRefs.current[sourceFile];
    if (el) {
      el.scrollIntoView({ behavior: 'smooth', block: 'start' });
    }
  };

  const toggleFileRules = async (sourceFile: string) => {
    if (expandedFiles.has(sourceFile)) {
      setExpandedFiles((prev) => {
        const n = new Set(prev);
        n.delete(sourceFile);
        return n;
      });
      return;
    }

    // Lazy load rules for this file
    if (!rulesMap.has(sourceFile)) {
      setLoadingRules((prev) => new Set(prev).add(sourceFile));
      try {
        const rules = await knowledgeApi.listRules(projectId, {
          source_file: sourceFile === '(sem arquivo)' ? undefined : sourceFile,
          limit: 500,
        });
        setRulesMap((prev) => new Map(prev).set(sourceFile, rules));
      } catch (error) {
        console.error('Failed to load rules for file:', error);
        setRulesMap((prev) => new Map(prev).set(sourceFile, []));
      } finally {
        setLoadingRules((prev) => {
          const n = new Set(prev);
          n.delete(sourceFile);
          return n;
        });
      }
    }
    setExpandedFiles((prev) => new Set(prev).add(sourceFile));
  };

  const getShortPath = (fullPath: string) => {
    if (fullPath === '(sem arquivo)') return fullPath;
    // Show last 2-3 segments
    const parts = fullPath.split('/');
    return parts.length > 3 ? '.../' + parts.slice(-3).join('/') : fullPath;
  };

  if (loading) {
    return (
      <Layout>
        <div className="flex items-center justify-center h-64">
          <div className="animate-spin rounded-full h-12 w-12 border-b-2 border-blue-600"></div>
        </div>
      </Layout>
    );
  }

  return (
    <Layout>
      <Breadcrumbs />

      <div className="flex gap-6">
        {/* Sidebar */}
        <div className="w-64 flex-shrink-0 hidden lg:block">
          <Card className="p-3 sticky top-4">
            <div className="flex items-center justify-between mb-3">
              <h2 className="text-xs font-semibold text-gray-500 uppercase tracking-wider">Arquivos</h2>
              <span className="text-xs text-gray-400">{totalRules} regras</span>
            </div>
            <div className="space-y-0.5 max-h-[70vh] overflow-y-auto">
              {fileEntries.map((entry) => (
                <button
                  key={entry.source_file}
                  onClick={() => scrollToFile(entry.source_file)}
                  className={`w-full text-left px-3 py-1.5 rounded text-sm transition-colors flex items-center justify-between gap-1 ${
                    activeFile === entry.source_file
                      ? 'bg-orange-100 text-orange-700 font-medium'
                      : 'text-gray-600 hover:bg-gray-100 hover:text-gray-900'
                  }`}
                  title={entry.source_file}
                >
                  <span className="truncate text-xs font-mono">{getShortPath(entry.source_file)}</span>
                  <span className={`text-xs px-1.5 py-0.5 rounded-full flex-shrink-0 ${
                    activeFile === entry.source_file ? 'bg-orange-200 text-orange-800' : 'bg-gray-200 text-gray-600'
                  }`}>
                    {entry.rules_count}
                  </span>
                </button>
              ))}
            </div>
          </Card>
        </div>

        {/* Main Content */}
        <div className="flex-1 min-w-0">
          {/* Page Header */}
          <div className="flex items-center justify-between mb-6">
            <div className="flex items-center gap-3">
              <button
                onClick={() => router.push(`/projects/${projectId}`)}
                className="text-gray-400 hover:text-gray-600"
              >
                <svg className="w-5 h-5" fill="none" stroke="currentColor" viewBox="0 0 24 24">
                  <path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M15 19l-7-7 7-7" />
                </svg>
              </button>
              <h1 className="text-2xl font-bold text-gray-900">Regras de Negocio</h1>
              <span className="px-2 py-0.5 rounded-full text-xs bg-orange-100 text-orange-700">{totalRules} regras</span>
            </div>
          </div>

          {/* Metadata */}
          <div className="text-xs text-gray-400 mb-4">
            {project?.name} — {fileEntries.length} arquivos-fonte
          </div>

          {/* Content - sections by source file */}
          {fileEntries.map((entry) => {
            const isExpanded = expandedFiles.has(entry.source_file);
            const isLoading = loadingRules.has(entry.source_file);
            const rules = rulesMap.get(entry.source_file) || [];

            return (
              <div
                key={entry.source_file}
                ref={(el) => { sectionRefs.current[entry.source_file] = el; }}
                className="mb-4"
              >
                <Card className="p-0 overflow-hidden">
                  {/* File header - clickable to expand */}
                  <button
                    onClick={() => toggleFileRules(entry.source_file)}
                    className="w-full px-4 py-3 bg-gray-50 border-b flex items-center justify-between hover:bg-gray-100 transition-colors"
                  >
                    <div className="flex items-center gap-2">
                      <svg
                        className={`w-3.5 h-3.5 text-gray-400 transition-transform ${isExpanded ? '' : '-rotate-90'}`}
                        fill="none" stroke="currentColor" viewBox="0 0 24 24"
                      >
                        <path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M19 9l-7 7-7-7" />
                      </svg>
                      <svg className="w-4 h-4 text-orange-500" fill="none" stroke="currentColor" viewBox="0 0 24 24">
                        <path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M9 12h6m-6 4h6m2 5H7a2 2 0 01-2-2V5a2 2 0 012-2h5.586a1 1 0 01.707.293l5.414 5.414a1 1 0 01.293.707V19a2 2 0 01-2 2z" />
                      </svg>
                      <span className="font-mono text-xs text-gray-700 text-left truncate">{entry.source_file}</span>
                    </div>
                    <span className="inline-flex items-center px-2 py-0.5 rounded-full text-xs font-medium bg-orange-100 text-orange-700 flex-shrink-0">
                      {entry.rules_count} {entry.rules_count === 1 ? 'regra' : 'regras'}
                    </span>
                  </button>

                  {/* Expanded rules */}
                  {isExpanded && (
                    isLoading ? (
                      <div className="px-4 py-3 text-sm text-gray-500">
                        Carregando regras...
                      </div>
                    ) : (
                      <div className="overflow-x-auto">
                        <table className="w-full">
                          <thead>
                            <tr className="border-b border-gray-100">
                              <th className="text-left py-2 px-4 text-xs font-medium text-gray-500">Regra</th>
                              <th className="text-left py-2 px-4 text-xs font-medium text-gray-500 w-28">Categoria</th>
                              <th className="text-left py-2 px-4 text-xs font-medium text-gray-500 w-28">Fonte</th>
                            </tr>
                          </thead>
                          <tbody>
                            {rules.map((rule) => (
                              <tr key={rule.id} className="border-b border-gray-50 hover:bg-gray-50">
                                <td className="py-2 px-4">
                                  {rule.title && (
                                    <div className="text-sm font-medium text-gray-800">{rule.title}</div>
                                  )}
                                  <div className="text-xs text-gray-500 line-clamp-2">{rule.description}</div>
                                </td>
                                <td className="py-2 px-4">
                                  <span className={`px-2 py-0.5 rounded text-xs font-medium ${CATEGORY_COLORS[rule.category] || CATEGORY_COLORS.general}`}>
                                    {rule.category}
                                  </span>
                                </td>
                                <td className="py-2 px-4">
                                  <span className="text-xs text-gray-500">{rule.source.replace(/_/g, ' ')}</span>
                                </td>
                              </tr>
                            ))}
                            {rules.length === 0 && (
                              <tr>
                                <td colSpan={3} className="py-3 px-4 text-sm text-gray-400 text-center">
                                  Nenhuma regra encontrada
                                </td>
                              </tr>
                            )}
                          </tbody>
                        </table>
                      </div>
                    )
                  )}
                </Card>
              </div>
            );
          })}
        </div>
      </div>
    </Layout>
  );
}
