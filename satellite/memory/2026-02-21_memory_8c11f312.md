"""
ORBIT Full Pipeline Script - PROMPT #244

Runs the complete RAG pipeline for the ORBIT project programmatically:
1. Scan all source files → rag_file_state
2. Process files → extract business rules → rag_documents
3. Generate hierarchical cards from business rules → tasks

Uses the same services as the UI but runs them directly.
"""
import asyncio
import logging
import os
import sys

# Add backend to path
backend_dir = os.path.dirname(os.path.dirname(os.path.abspath(__file__)))
sys.path.insert(0, backend_dir)

# Load .env BEFORE any imports that use os.getenv
from dotenv import load_dotenv
env_path = os.path.join(backend_dir, ".env")
load_dotenv(env_path, override=True)
logger_init = logging.getLogger("orbit_pipeline.init")
logger_init.info(f"Loaded .env from {env_path}")
logger_init.info(f"OLLAMA_HOST={os.getenv('OLLAMA_HOST')}")
logger_init.info(f"DATABASE_URL={os.getenv('DATABASE_URL')}")

logging.basicConfig(
    level=logging.INFO,
    format="%(asctime)s [%(levelname)s] %(name)s: %(message)s",
)
logger = logging.getLogger("orbit_pipeline")

# Suppress noisy loggers
logging.getLogger("httpx").setLevel(logging.WARNING)
logging.getLogger("httpcore").setLevel(logging.WARNING)


async def main():
    from app.database import SessionLocal
    from app.models.project import Project

    db = SessionLocal()

    try:
        # Find ORBIT project
        project = db.query(Project).filter(Project.code_path == "/home/igorhaf/orbit").first()
        if not project:
            logger.error("ORBIT project not found!")
            return

        project_id = project.id
        logger.info(f"Found ORBIT project: {project.name} ({project_id})")
        logger.info(f"  code_path: {project.code_path}")
        logger.info(f"  initial_scan_complete: {project.initial_scan_complete}")

        # =====================================================================
        # PHASE 1: Scan files and detect changes
        # =====================================================================
        logger.info("=" * 60)
        logger.info("PHASE 1: Scanning source files...")
        logger.info("=" * 60)

        from app.services.continuous_rag_service import ContinuousRAGService

        rag_service = ContinuousRAGService(db)
        scan_result = await rag_service.scan_for_changes(project_id)

        logger.info(f"Scan result: {scan_result}")
        logger.info(f"  New files: {scan_result.get('new_files', 0)}")
        logger.info(f"  Modified: {scan_result.get('modified_files', 0)}")
        logger.info(f"  Total tracked: {scan_result.get('total_tracked', 0)}")

        # =====================================================================
        # PHASE 2: Process pending files (extract business rules via AI)
        # =====================================================================
        logger.info("=" * 60)
        logger.info("PHASE 2: Extracting business rules from code via AI...")
        logger.info("=" * 60)

        total_processed = 0
        total_rules = 0
        total_failed = 0
        batch_num = 0
        batch_size = 20  # Process 20 files per batch

        while True:
            batch_num += 1
            logger.info(f"--- Batch {batch_num} (size={batch_size}) ---")
            process_result = await rag_service.process_pending_files(
                project_id, batch_size=batch_size
            )

            files_processed = process_result.get('files_processed', 0)
            rules_extracted = process_result.get('total_rules_extracted', 0)
            files_failed = process_result.get('files_failed', 0)

            total_processed += files_processed
            total_rules += rules_extracted
            total_failed += files_failed

            logger.info(f"  Batch {batch_num}: {files_processed} processed, {rules_extracted} rules, {files_failed} failed")
            logger.info(f"  Running total: {total_processed} processed, {total_rules} rules")

            # Stop when no more pending files
            if files_processed == 0:
                logger.info("No more pending files to process")
                break

        logger.info(f"Phase 2 complete:")
        logger.info(f"  Total files processed: {total_processed}")
        logger.info(f"  Total rules extracted: {total_rules}")
        logger.info(f"  Total failed: {total_failed}")

        # =====================================================================
        # PHASE 3: Generate hierarchical cards from business rules
        # =====================================================================
        logger.info("=" * 60)
        logger.info("PHASE 3: Generating hierarchical cards from business rules...")
        logger.info("=" * 60)

        from app.services.context_generator.service import ContextGeneratorService

        ctx_service = ContextGeneratorService(db)
        cards = await ctx_service.generate_business_rule_cards(project_id)

        logger.info(f"Cards generated: {len(cards)}")

        # Count by type
        type_counts = {}
        for card in cards:
            item_type = card.get("item_type", "unknown")
            type_counts[item_type] = type_counts.get(item_type, 0) + 1

        for item_type, count in sorted(type_counts.items()):
            logger.info(f"  {item_type}: {count}")

        # =====================================================================
        # Summary
        # =====================================================================
        logger.info("=" * 60)
        logger.info("PIPELINE COMPLETE!")
        logger.info("=" * 60)

        from sqlalchemy import text

        rag_count = db.execute(text(
            "SELECT COUNT(*) FROM rag_documents WHERE project_id = :pid"
        ), {"pid": str(project_id)}).scalar()

        rule_count = db.execute(text(
            "SELECT COUNT(*) FROM rag_documents WHERE project_id = :pid "
            "AND (metadata->>'type' = 'business_rule' OR metadata->>'content_type' = 'business_rule')"
        ), {"pid": str(project_id)}).scalar()

        task_count = db.execute(text(
            "SELECT item_type, COUNT(*) FROM tasks WHERE project_id = :pid GROUP BY item_type ORDER BY item_type"
        ), {"pid": str(project_id)}).fetchall()

        file_count = db.execute(text(
            "SELECT status, COUNT(*) FROM rag_file_state WHERE project_id = :pid GROUP BY status"
        ), {"pid": str(project_id)}).fetchall()

        logger.info(f"RAG Documents: {rag_count} total, {rule_count} business rules")
        logger.info(f"File states: {dict(file_count)}")
        logger.info(f"Cards: {dict(task_count)}")

    except Exception as e:
        logger.error(f"Pipeline failed: {e}", exc_info=True)
        raise
    finally:
        db.close()


if __name__ == "__main__":
    asyncio.run(main())
