<?php

namespace App\Http\Controllers\Instrutor;

use App\Http\Controllers\Controller;
use App\Models\Course;
use App\Models\CourseReview;
use Illuminate\Http\Request;

class AvaliacoesRecebidasController extends Controller
{
    public function index(Request $request)
    {
        $user = $request->user();
        if (!$user || $user->role !== 'instrutor') {
            abort(403);
        }

        $courses = Course::where('instructor_id', $user->id)->pluck('id');
        
        $avaliacoes = CourseReview::whereIn('course_id', $courses)
            ->with(['course', 'user'])
            ->latest()
            ->get()
            ->map(function (CourseReview $review) {
                return [
                    'id' => $review->id,
                    'curso' => [
                        'id' => $review->course->id,
                        'titulo' => $review->course->title,
                        'thumb' => $review->course->cover_url,
                    ],
                    'aluno' => [
                        'nome' => $review->user?->name ?? 'Usuário',
                        'avatar' => $review->user?->avatar_url,
                    ],
                    'nota' => $review->rating,
                    'comentario' => $review->comment,
                    'data' => $review->created_at->toDateString(),
                    'tempo' => $review->created_at->diffForHumans(),
                    'status' => $review->rating >= 4 ? 'positiva' : ($review->rating <= 2 ? 'negativa' : 'neutra'),
                ];
            })
            ->all();

        $resumo = [
            'total' => count($avaliacoes),
            'positivas' => collect($avaliacoes)->where('status', 'positiva')->count(),
            'neutras' => collect($avaliacoes)->where('status', 'neutra')->count(),
            'negativas' => collect($avaliacoes)->where('status', 'negativa')->count(),
            'media' => CourseReview::whereIn('course_id', $courses)->avg('rating') ?: 0,
        ];

        return view('instrutor.avaliacoes-recebidas', compact('avaliacoes', 'resumo'));
    }
}
