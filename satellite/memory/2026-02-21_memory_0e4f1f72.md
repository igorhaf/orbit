<?php

namespace App\Http\Controllers\Aluno;

use App\Http\Controllers\Controller;
use App\Models\Subscription;
use Illuminate\Http\Request;

class PlanoController extends Controller
{
	public function show(Request $request)
	{
		$user = $request->user();
		$sub = Subscription::where('user_id', $user->id)->first();
		$plano = $sub ? [
			'nome' => $sub->plan_name,
			'ciclo' => ucfirst($sub->cycle),
			'proximo' => optional($sub->next_due_at)?->toDateString(),
			'status' => $sub->status,
			'beneficios' => $sub->benefits ?? [],
		] : [
			'nome' => 'Free', 'ciclo' => '-', 'proximo' => null, 'status' => 'inativo', 'beneficios' => []
		];
		return view('aluno.meu-plano', compact('plano'));
	}
}
