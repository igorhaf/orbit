"""
Symbol Extractor - PROMPT #230

Regex-based code symbol extraction for architectural summarization.
Extracts class names, function signatures, imports, constants, and
relationships WITHOUT using AI or AST parsing.

Supports: Python, JS/TS, PHP, Java, Go, Ruby, C#, Kotlin, Swift.

Purpose: Replace raw code chunks in memory scan with compact symbol maps
that are 5-10x smaller but preserve architectural understanding.
"""

import re
import logging
from typing import Dict, List, Optional, Set
from pathlib import Path

logger = logging.getLogger(__name__)


# ---------------------------------------------------------------------------
# Language-specific regex patterns
# ---------------------------------------------------------------------------

IMPORT_PATTERNS = {
    "python": [
        re.compile(r"^(?:from\s+(\S+)\s+)?import\s+(.+)$", re.MULTILINE),
    ],
    "javascript": [
        re.compile(r"^import\s+.*?from\s+['\"](.+?)['\"]", re.MULTILINE),
        re.compile(r"(?:const|let|var)\s+.*?=\s*require\(['\"](.+?)['\"]\)", re.MULTILINE),
    ],
    "php": [
        re.compile(r"^use\s+(.+?);", re.MULTILINE),
        re.compile(r"^require(?:_once)?\s+['\"](.+?)['\"]", re.MULTILINE),
    ],
    "java": [
        re.compile(r"^import\s+(.+?);", re.MULTILINE),
    ],
    "go": [
        re.compile(r'"(.+?)"', re.MULTILINE),  # inside import block
    ],
    "ruby": [
        re.compile(r"^require\s+['\"](.+?)['\"]", re.MULTILINE),
    ],
    "csharp": [
        re.compile(r"^using\s+(.+?);", re.MULTILINE),
    ],
}

CLASS_PATTERN = re.compile(
    r"^(?:export\s+)?(?:abstract\s+)?(?:public\s+|private\s+|protected\s+)?"
    r"(?:final\s+)?(?:data\s+)?(?:class|struct|interface|trait|enum)\s+"
    r"(\w+)",
    re.MULTILINE,
)

FUNCTION_PATTERNS = [
    # Python: def func_name(args):
    re.compile(r"^\s*(?:async\s+)?def\s+(\w+)\s*\(([^)]*)\)", re.MULTILINE),
    # JS/TS: function name(args) / async function name(args)
    re.compile(r"^\s*(?:export\s+)?(?:async\s+)?function\s+(\w+)\s*\(([^)]*)\)", re.MULTILINE),
    # PHP: public function name(args)
    re.compile(r"^\s*(?:public|private|protected|static)\s+(?:static\s+)?function\s+(\w+)\s*\(([^)]*)\)", re.MULTILINE),
    # Go: func (receiver) Name(args) returnType
    re.compile(r"^\s*func\s+(?:\([^)]*\)\s+)?(\w+)\s*\(([^)]*)\)", re.MULTILINE),
    # Java/C#/Kotlin: public ReturnType methodName(args)
    re.compile(r"^\s*(?:public|private|protected|internal|override|suspend)\s+(?:static\s+)?(?:\w+\s+)?(\w+)\s*\(([^)]*)\)", re.MULTILINE),
    # Ruby: def method_name(args)
    re.compile(r"^\s*def\s+(?:self\.)?(\w+)(?:\s*\(([^)]*)\))?", re.MULTILINE),
    # Arrow functions assigned: const name = (args) =>
    re.compile(r"^\s*(?:export\s+)?(?:const|let|var)\s+(\w+)\s*=\s*(?:async\s+)?\(([^)]*)\)\s*=>", re.MULTILINE),
]

CONSTANT_PATTERNS = [
    # Python: CONST_NAME = value
    re.compile(r"^([A-Z][A-Z0-9_]{2,})\s*=\s*(.+?)$", re.MULTILINE),
    # JS/TS: const CONST_NAME = value
    re.compile(r"^\s*(?:export\s+)?const\s+([A-Z][A-Z0-9_]{2,})\s*=", re.MULTILINE),
    # PHP: const CONST_NAME = value / define('CONST', value)
    re.compile(r"^\s*(?:public\s+)?const\s+([A-Z][A-Z0-9_]{2,})\s*=", re.MULTILINE),
]

DECORATOR_PATTERNS = [
    # Python decorators: @decorator
    re.compile(r"^\s*@(\w+(?:\.\w+)*)", re.MULTILINE),
    # Java/Kotlin annotations: @Annotation
    re.compile(r"^\s*@(\w+)(?:\(.*?\))?$", re.MULTILINE),
    # PHP attributes: #[Attribute]
    re.compile(r"^\s*#\[(\w+)", re.MULTILINE),
]

# Patterns indicating business logic (for scoring)
BUSINESS_LOGIC_INDICATORS = [
    re.compile(r"(?:validate|validation|validator)\w*", re.IGNORECASE),
    re.compile(r"(?:calculate|compute|formula)\w*", re.IGNORECASE),
    re.compile(r"(?:permission|authorize|guard|policy)\w*", re.IGNORECASE),
    re.compile(r"(?:if|when|unless|switch)\s+.*?(?:status|state|role|type)\b", re.IGNORECASE),
    re.compile(r"(?:raise|throw)\s+\w*(?:Error|Exception)", re.IGNORECASE),
    re.compile(r"(?:min|max|limit|threshold|timeout|retry)\s*[=<>]", re.IGNORECASE),
]

# Extension to language mapping
EXT_TO_LANGUAGE = {
    ".py": "python",
    ".js": "javascript", ".ts": "javascript", ".tsx": "javascript", ".jsx": "javascript",
    ".php": "php", ".blade.php": "php",
    ".java": "java", ".kt": "java",
    ".go": "go",
    ".rb": "ruby",
    ".cs": "csharp",
    ".swift": "java",  # similar enough patterns
    ".vue": "javascript", ".svelte": "javascript",
}


def detect_language(filename: str) -> str:
    """Detect language from filename extension."""
    path = Path(filename)
    ext = path.suffix.lower()
    # Handle compound extensions like .blade.php
    if filename.endswith(".blade.php"):
        return "php"
    return EXT_TO_LANGUAGE.get(ext, "unknown")


def extract_symbols(filename: str, content: str) -> Dict:
    """
    Extract symbols from a single code file.

    Returns:
        {
            "filename": str,
            "language": str,
            "lines": int,
            "imports": [str],
            "classes": [str],
            "functions": [{"name": str, "params": str}],
            "constants": [str],
            "decorators": [str],
            "business_logic_lines": [str],
            "relationships": [str],
        }
    """
    language = detect_language(filename)
    lines = content.split("\n")
    line_count = len(lines)

    result = {
        "filename": filename,
        "language": language,
        "lines": line_count,
        "imports": [],
        "classes": [],
        "functions": [],
        "constants": [],
        "decorators": [],
        "business_logic_lines": [],
        "relationships": [],
    }

    # Extract imports
    import_pats = IMPORT_PATTERNS.get(language, [])
    seen_imports: Set[str] = set()
    for pat in import_pats:
        for match in pat.finditer(content):
            imp = match.group(1) or match.group(0)
            imp = imp.strip().rstrip(";")
            if imp and imp not in seen_imports:
                seen_imports.add(imp)
                result["imports"].append(imp)

    # Extract classes
    for match in CLASS_PATTERN.finditer(content):
        class_name = match.group(1)
        if class_name and not class_name.startswith("_"):
            result["classes"].append(class_name)

    # Extract functions
    seen_funcs: Set[str] = set()
    for pat in FUNCTION_PATTERNS:
        for match in pat.finditer(content):
            func_name = match.group(1)
            params = match.group(2).strip() if match.lastindex >= 2 and match.group(2) else ""
            if func_name and func_name not in seen_funcs:
                # Skip dunder methods and trivial getters
                if func_name.startswith("__") and func_name.endswith("__"):
                    if func_name not in ("__init__", "__call__", "__enter__", "__exit__"):
                        continue
                seen_funcs.add(func_name)
                # Truncate long param lists
                if len(params) > 80:
                    params = params[:77] + "..."
                result["functions"].append({"name": func_name, "params": params})

    # Extract constants
    seen_consts: Set[str] = set()
    for pat in CONSTANT_PATTERNS:
        for match in pat.finditer(content):
            const_name = match.group(1)
            if const_name and const_name not in seen_consts:
                seen_consts.add(const_name)
                result["constants"].append(const_name)

    # Extract decorators/annotations
    seen_decorators: Set[str] = set()
    for pat in DECORATOR_PATTERNS:
        for match in pat.finditer(content):
            dec = match.group(1)
            if dec and dec not in seen_decorators:
                seen_decorators.add(dec)
                result["decorators"].append(dec)

    # Find business logic indicator lines
    for i, line in enumerate(lines):
        stripped = line.strip()
        if not stripped or stripped.startswith(("#", "//", "/*", "*", "--")):
            continue
        for biz_pat in BUSINESS_LOGIC_INDICATORS:
            if biz_pat.search(stripped):
                # Keep a compact version of the line
                compact = stripped[:120]
                result["business_logic_lines"].append(f"L{i+1}: {compact}")
                break

    # Limit business logic lines to top 10
    result["business_logic_lines"] = result["business_logic_lines"][:10]

    # Build relationships from imports
    for imp in result["imports"]:
        # Extract module/file references
        parts = imp.replace("\\", "/").split("/")
        if len(parts) > 1:
            result["relationships"].append(f"imports {parts[-1]}")
        else:
            result["relationships"].append(f"uses {imp.split('.')[-1]}")

    # Deduplicate relationships
    result["relationships"] = list(dict.fromkeys(result["relationships"]))[:15]

    return result


def format_symbol_map(symbols: Dict) -> str:
    """
    Format extracted symbols as a compact text representation for LLM analysis.

    This is typically 5-10x smaller than the raw code while preserving
    architectural understanding.
    """
    parts = []
    parts.append(f"FILE: {symbols['filename']} ({symbols['language']}, {symbols['lines']} lines)")

    if symbols["classes"]:
        parts.append(f"  CLASSES: {', '.join(symbols['classes'])}")

    if symbols["functions"]:
        func_strs = []
        for f in symbols["functions"][:20]:  # Max 20 functions
            if f["params"]:
                func_strs.append(f"{f['name']}({f['params']})")
            else:
                func_strs.append(f"{f['name']}()")
        parts.append(f"  FUNCTIONS: {', '.join(func_strs)}")

    if symbols["constants"]:
        parts.append(f"  CONSTANTS: {', '.join(symbols['constants'][:10])}")

    if symbols["decorators"]:
        parts.append(f"  ANNOTATIONS: {', '.join(symbols['decorators'][:10])}")

    if symbols["imports"]:
        parts.append(f"  IMPORTS: {', '.join(symbols['imports'][:15])}")

    if symbols["business_logic_lines"]:
        parts.append("  BUSINESS LOGIC:")
        for line in symbols["business_logic_lines"]:
            parts.append(f"    {line}")

    if symbols["relationships"]:
        parts.append(f"  RELATIONSHIPS: {', '.join(symbols['relationships'][:10])}")

    return "\n".join(parts)


def extract_and_format_batch(samples: List[Dict[str, str]]) -> str:
    """
    Extract symbols from a batch of code samples and format as compact text.

    Args:
        samples: List of {"filename": str, "content": str, "type": str}

    Returns:
        Compact architectural summary text
    """
    parts = []
    total_original_chars = 0
    total_compressed_chars = 0

    for sample in samples:
        filename = sample.get("filename", "unknown")
        content = sample.get("content", "")
        total_original_chars += len(content)

        try:
            symbols = extract_symbols(filename, content)
            formatted = format_symbol_map(symbols)
            total_compressed_chars += len(formatted)
            parts.append(formatted)
        except Exception as e:
            logger.warning(f"Symbol extraction failed for {filename}: {e}")
            # Fallback: include first 500 chars of content
            fallback = f"FILE: {filename} (extraction failed)\n  PREVIEW: {content[:500]}"
            total_compressed_chars += len(fallback)
            parts.append(fallback)

    compression_ratio = (
        round(total_original_chars / max(total_compressed_chars, 1), 1)
        if total_original_chars > 0 else 1.0
    )
    logger.info(
        f"Symbol extraction: {len(samples)} files, "
        f"{total_original_chars} -> {total_compressed_chars} chars "
        f"({compression_ratio}x compression)"
    )

    return "\n\n".join(parts)
