<?php

namespace App\Http\Controllers\Ajuda;

use App\Http\Controllers\Controller;
use App\Models\Faq;
use App\Models\FaqCategory;
use Illuminate\Http\Request;
use Illuminate\View\View;

class CentralController extends Controller
{
    public function index(): View
    {
        $categories = FaqCategory::query()
            ->active()
            ->ordered()
            ->withCount(['faqs' => fn ($q) => $q->published()])
            ->get();

        $popular = Faq::query()
            ->published()
            ->popular()
            ->with('category')
            ->take(5)
            ->get();

        return view('ajuda.index', compact('categories', 'popular'));
    }

    public function show(string $slug): View
    {
        $faq = Faq::query()
            ->published()
            ->where('slug', $slug)
            ->with('category')
            ->firstOrFail();

        $faq->increment('views_count');

        $related = Faq::query()
            ->published()
            ->where('faq_category_id', $faq->faq_category_id)
            ->where('id', '!=', $faq->id)
            ->orderByDesc('views_count')
            ->take(4)
            ->get();

        return view('ajuda.show', compact('faq', 'related'));
    }

    public function categoria(string $slug): View
    {
        $category = FaqCategory::query()
            ->active()
            ->where('slug', $slug)
            ->firstOrFail();

        $faqs = Faq::query()
            ->published()
            ->where('faq_category_id', $category->id)
            ->orderBy('sort_order')
            ->get();

        $categories = FaqCategory::query()
            ->active()
            ->ordered()
            ->withCount(['faqs' => fn ($q) => $q->published()])
            ->get();

        return view('ajuda.categoria', compact('category', 'faqs', 'categories'));
    }

    public function search(Request $request): View
    {
        $term = $request->query('q', '');

        $results = collect();

        if (strlen($term) >= 2) {
            $results = Faq::query()
                ->published()
                ->search($term)
                ->with('category')
                ->orderByDesc('views_count')
                ->take(20)
                ->get();
        }

        return view('ajuda.busca', compact('term', 'results'));
    }
}
