<?php

namespace App\Http\Controllers\Aulas;

use App\Http\Controllers\Controller;
use App\Models\Lesson;
use App\Models\LessonDiscussion;
use Illuminate\Http\Request;

class DiscussoesFormController extends Controller
{
	public function create(Request $request, string $id)
	{
		$lesson = $this->findLesson($id);
		abort_unless($lesson, 404);
		$course = $lesson->module->course;

		$curso = [ 'id' => $course->id, 'titulo' => $course->title ];
		$licao = [ 'id' => $lesson->id, 'titulo' => $lesson->title ];
		$todasTags = LessonDiscussion::where('lesson_id', $lesson->id)->pluck('tags')->flatten(1)->filter()->unique()->values()->all();
		$draft = [ 'titulo' => '', 'corpo' => '', 'tags' => [], 'duvida' => false, 'anexos' => [] ];
		$isEdit = false;

		return view('aulas.discussoes-form', compact('curso','licao','todasTags','draft','isEdit'));
	}

	public function edit(Request $request, string $id, string $postId)
	{
		$lesson = $this->findLesson($id);
		abort_unless($lesson, 404);
		$course = $lesson->module->course;
		$post = LessonDiscussion::where('lesson_id', $lesson->id)->findOrFail($postId);

		$curso = [ 'id' => $course->id, 'titulo' => $course->title ];
		$licao = [ 'id' => $lesson->id, 'titulo' => $lesson->title ];
		$todasTags = LessonDiscussion::where('lesson_id', $lesson->id)->pluck('tags')->flatten(1)->filter()->unique()->values()->all();
		$draft = [
			'titulo' => $post->title,
			'corpo' => $post->body,
			'tags' => (array) ($post->tags ?? []),
			'duvida' => (bool) $post->is_question,
			'anexos' => [],
		];
		$isEdit = true;

		return view('aulas.discussoes-form', compact('curso','licao','todasTags','draft','isEdit'));
	}

	private function findLesson(string $id): ?Lesson
	{
		if (ctype_digit($id)) return Lesson::with('module.course')->find($id);
		return Lesson::with('module.course')->where('slug', $id)->first();
	}
}
