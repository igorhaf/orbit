<?php

namespace App\Http\Controllers\Categorias;

use App\Http\Controllers\Controller;
use App\Models\Course;
use Illuminate\Http\Request;
use Illuminate\Support\Str;

class ExplorarController extends Controller
{
	public function show(Request $request, string $id)
	{
		$tagSlug = $id;
		$nome = Str::of($tagSlug)->replace('-', ' ')->title()->toString();
		$courses = Course::query()->published()->with('instructor')
			->whereJsonContains('tags', $nome)
			->orWhereJsonContains('tags', Str::of($tagSlug)->replace('-', ' ')->toString())
			->get();

		$categoria = ['nome'=>$nome,'area'=> $courses->pluck('category')->filter()->first() ?? 'Geral'];

		// Subcategorias simples: a partir de palavras-chave comuns
		$keywords = ['Fundamentos'=>'📦','Segurança'=>'🔐','Testes'=>'🧪','Performance'=>'⚙️','Integrações'=>'🧩'];
		$subcategorias = [];
		foreach ($keywords as $k => $icon) {
			$subcategorias[] = ['id'=> Str::slug($k), 'icon'=> $icon, 'nome'=> $k];
		}

		$cursosPorSub = [];
		foreach ($subcategorias as $sub) {
			$k = $sub['nome'];
			$subset = $courses->filter(function($c) use ($k){
				$hay = strtolower(($c->title ?? '').' '.implode(' ', (array)($c->tags ?? [])));
				return str_contains($hay, strtolower($k));
			});
			$cursosPorSub[$sub['id']] = $subset->map(function(Course $c){
				return [
					'id' => $c->id,
					'titulo' => $c->title,
					'duracao' => (int)($c->duration_minutes ?? 0),
					'nivel' => $c->level ?? 'Intermediário',
					'idioma' => $c->language === 'pt' ? 'Português' : strtoupper($c->language),
					'rating' => (float)($c->rating_average ?? 0),
					'ratings' => (int)($c->rating_count ?? 0),
					'thumb' => $c->cover_url,
					'preco' => ((int)($c->price_cents ?? 0))/100,
				];
			})->values()->all();
		}

		return view('categorias.explorar', compact('categoria','subcategorias','cursosPorSub','categoryId'));
	}
}
