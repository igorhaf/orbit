<?php

namespace App\Http\Controllers\Instrutor;

use App\Http\Controllers\Controller;
use App\Models\Course;
use App\Models\Certificate;
use Illuminate\Http\Request;

class CertificadosController extends Controller
{
    public function show(Request $request, string $id)
    {
        $user = $request->user();
        if (!$user || $user->role !== 'instrutor') {
            abort(403);
        }

        $course = Course::where('id', $id)
            ->where('instructor_id', $user->id)
            ->firstOrFail();

        $curso = [
            'id' => $course->id,
            'titulo' => $course->title,
            'cover' => $course->cover_url,
        ];

        // Buscar certificados emitidos para este curso
        $certificados = Certificate::where('course_id', $course->id)
            ->with('user')
            ->latest()
            ->get()
            ->map(function (Certificate $certificate) {
                return [
                    'id' => $certificate->id,
                    'codigo' => $certificate->code,
                    'aluno' => [
                        'nome' => $certificate->user?->name ?? 'Usuário',
                        'email' => $certificate->user?->email,
                    ],
                    'data_emissao' => $certificate->issued_at?->toDateString() ?? $certificate->created_at->toDateString(),
                    'status' => $certificate->status ?? 'ativo',
                    'download_url' => $certificate->download_url,
                ];
            })
            ->all();

        $resumo = [
            'total' => count($certificados),
            'ativos' => collect($certificados)->where('status', 'ativo')->count(),
            'mes_atual' => collect($certificados)->filter(function ($cert) {
                return now()->startOfMonth()->lte($cert['data_emissao']) && 
                       now()->endOfMonth()->gte($cert['data_emissao']);
            })->count(),
        ];

        return view('instrutor.cursos.certificados', compact('curso', 'certificados', 'resumo'));
    }
}
