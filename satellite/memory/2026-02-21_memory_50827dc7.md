<?php

namespace App\Http\Controllers\Instrutor;

use App\Http\Controllers\Controller;
use App\Models\Course;
use App\Models\Enrollment;
use Illuminate\Http\Request;

class RepasseDetalheController extends Controller
{
    public function show(Request $request, string $id)
    {
        $user = $request->user();
        if (!$user || $user->role !== 'instrutor') {
            abort(403);
        }

        // Buscar curso do instrutor
        $course = Course::where('id', $id)
            ->where('instructor_id', $user->id)
            ->firstOrFail();

        // Calcular repasse
        $enrollments = $course->enrollments()
            ->where('status', 'ativo')
            ->where('created_at', '>=', now()->subDays(30))
            ->get();

        $receitaBruta = $enrollments->sum(function ($enrollment) use ($course) {
            return $course->price_cents ? $course->price_cents / 100 : 0;
        });

        $comissao = $receitaBruta * 0.7; // 70% para o instrutor
        $taxaPlataforma = $receitaBruta * 0.3; // 30% para a plataforma

        $repasse = [
            'id' => $course->id,
            'curso' => [
                'titulo' => $course->title,
                'thumb' => $course->cover_url,
                'preco' => $course->price_cents ? $course->price_cents / 100 : 0,
            ],
            'periodo' => 'Janeiro 2025',
            'receitaBruta' => $receitaBruta,
            'comissao' => $comissao,
            'taxaPlataforma' => $taxaPlataforma,
            'inscricoes' => $enrollments->count(),
            'status' => 'Processado',
            'dataProcessamento' => now()->subDays(2)->toDateString(),
        ];

        return view('instrutor.repasse-detalhe', compact('repasse'));
    }
}
