<?php

namespace App\Http\Controllers\Cursos;

use App\Http\Controllers\Controller;
use App\Models\Course;
use Illuminate\Http\Request;

class CatalogoController extends Controller
{
	public function index(Request $request)
	{
		$q = (string) $request->query('q');

		$published = Course::query()->published();

		$courses = (clone $published)
			->when($q, fn($qBuilder) => $qBuilder->where('title', 'like', '%'.$q.'%'))
			->latest('published_at')
			->limit(120)
			->get(['id','title','category','level','language','duration_minutes','cover_url','published_at','rating_average','rating_count','price_cents','tags']);

		$mapped = $courses->map(function (Course $course) {
			return [
				'id' => $course->id,
				'title' => $course->title,
				'category' => $course->category ?? 'Programação',
				'level' => $course->level ?? 'Iniciante',
				'language' => $this->mapLanguage($course->language),
				'duration' => (int) ($course->duration_minutes ?? 0),
				'rating' => (float) ($course->rating_average ?: 0),
				'ratings' => (int) ($course->rating_count ?: 0),
				'price' => (float) (($course->price_cents ?? 0) / 100),
				'thumb' => $course->cover_url ?: 'https://images.unsplash.com/photo-1517433456452-f9633a875f6f?q=80&w=1600&auto=format&fit=crop',
				'createdAt' => optional($course->published_at)->toDateString() ?? now()->toDateString(),
				'tags' => (array) ($course->tags ?? []),
			];
		});

		$categorias = array_values(array_unique(array_merge(
			$courses->pluck('category')->filter()->values()->all(),
			$courses->pluck('tags')->flatten(1)->filter()->values()->all(),
		)));
		sort($categorias, SORT_FLAG_CASE | SORT_NATURAL);

		$niveis = array_values(array_unique($courses->pluck('level')->filter()->values()->all() ?: ['Iniciante','Intermediário','Avançado']));
		$idiomas = ['Português','Inglês','Espanhol'];

		return view('cursos.index', [
			'courses' => $mapped,
			'categorias' => $categorias,
			'niveis' => $niveis,
			'idiomas' => $idiomas,
		]);
	}

	private function mapLanguage(?string $code): string
	{
		return match ($code) {
			'en' => 'Inglês',
			'es' => 'Espanhol',
			default => 'Português',
		};
	}
}
