<?php

namespace App\Http\Controllers\Aulas;

use App\Http\Controllers\Controller;
use App\Models\Lesson;
use App\Models\LessonDiscussion;
use Illuminate\Http\Request;

class DiscussoesMinhasController extends Controller
{
	public function index(Request $request, string $id)
	{
		$lesson = $this->findLesson($id);
		abort_unless($lesson, 404);
		$course = $lesson->module->course;

		$curso = [ 'id' => $course->id, 'titulo' => $course->title ];
		$licao = [ 'id' => $lesson->id, 'titulo' => $lesson->title ];
		$userId = optional($request->user())->id;
		$mine = $userId ? LessonDiscussion::where('lesson_id', $lesson->id)->where('user_id', $userId)->latest()->get() : collect();
		$interacoes = $mine->map(function($d) use ($curso, $licao){
			return [
				'tipo' => 'post',
				'topico' => $d->title,
				'curso' => $curso['titulo'],
				'aula' => $licao['titulo'],
				'data' => optional($d->created_at)->toDateString(),
				'status' => $d->resolved_at ? 'resolvido' : 'aberto',
				'votos' => (int) ($d->votes_count ?? 0),
				'respostas' => 0,
				'id' => (string) $d->id,
				'seguindo' => false,
			];
		})->values()->all();

		return view('aulas.discussoes-minhas', compact('curso','licao','interacoes'));
	}

	private function findLesson(string $id): ?Lesson
	{
		if (ctype_digit($id)) return Lesson::with('module.course')->find($id);
		return Lesson::with('module.course')->where('slug', $id)->first();
	}
}
