<?php

namespace App\Http\Controllers\Instrutor;

use App\Http\Controllers\Controller;
use App\Models\Course;
use App\Models\LessonMaterial;
use Illuminate\Http\Request;

class MateriaisController extends Controller
{
    public function show(Request $request, string $id)
    {
        $user = $request->user();
        if (!$user || $user->role !== 'instrutor') {
            abort(403);
        }

        $course = Course::where('id', $id)
            ->where('instructor_id', $user->id)
            ->with(['modules.lessons.materials'])
            ->firstOrFail();

        $curso = [
            'id' => $course->id,
            'titulo' => $course->title,
            'cover' => $course->cover_url,
        ];

        $materiais = collect();
        
        foreach ($course->modules as $module) {
            foreach ($module->lessons as $lesson) {
                foreach ($lesson->materials as $material) {
                    $materiais->push([
                        'id' => $material->id,
                        'nome' => $material->name,
                        'tipo' => $material->type,
                        'tamanho' => $material->size_kb ? round($material->size_kb / 1024, 2) . ' MB' : '—',
                        'url' => $material->url,
                        'licao' => [
                            'id' => $lesson->id,
                            'titulo' => $lesson->title,
                        ],
                        'modulo' => [
                            'id' => $module->id,
                            'titulo' => $module->title,
                        ],
                        'atualizado' => $material->updated_at->toDateString(),
                        'tags' => $material->tags ?? [],
                    ]);
                }
            }
        }

        $materiais = $materiais->sortBy('modulo.titulo')->values()->all();

        $resumo = [
            'total' => count($materiais),
            'por_tipo' => collect($materiais)->groupBy('tipo')->map->count()->all(),
            'tamanho_total' => collect($materiais)->sum(function ($m) {
                return is_numeric(str_replace(' MB', '', $m['tamanho'])) ? (float) str_replace(' MB', '', $m['tamanho']) : 0;
            }),
        ];

        return view('instrutor.cursos.materiais', compact('curso', 'materiais', 'resumo'));
    }
}
