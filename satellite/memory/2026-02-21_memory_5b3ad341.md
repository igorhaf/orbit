<?php

namespace App\Http\Controllers\Cursos;

use App\Http\Controllers\Controller;
use App\Models\Course;
use Illuminate\Http\Request;

class ProgressController extends Controller
{
	public function show(Request $request, string $id)
	{
		$course = $this->findCourse($id);
		abort_unless($course, 404);

		$curso = [ 'id' => $course->id, 'titulo' => $course->title ];
		$modulos = $course->modules()->with('lessons')->orderBy('position')->get()->map(function($m){
			return [
				'id' => $m->id,
				'titulo' => $m->title,
				'aulas' => $m->lessons->map(fn($l)=> [ 'id'=>$l->id, 'titulo'=>$l->title, 'duracao'=>$l->duration_minutes ])->all(),
			];
		})->all();
		$badges = [];

		return view('cursos.progresso', compact('curso','modulos','badges'));
	}

	private function findCourse(string $id): ?Course
	{
		if (ctype_digit($id)) return Course::find($id);
		return Course::where('slug', $id)->first();
	}
}
