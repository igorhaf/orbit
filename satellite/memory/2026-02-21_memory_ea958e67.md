<?php

namespace App\Http\Controllers\Aluno;

use App\Http\Controllers\Controller;
use App\Models\Course;
use App\Models\Enrollment;
use App\Models\LessonMaterial;
use Illuminate\Http\Request;

class CursoAndamentoController extends Controller
{
	public function show(Request $request, string $id)
	{
		$course = $this->findCourse($id);
		abort_unless($course, 404);

		$course->load(['modules.lessons' => function($q){ $q->orderBy('position'); }]);

		$user = $request->user();
		$enrollment = $user ? Enrollment::where('user_id', $user->id)->where('course_id', $course->id)->first() : null;

		$allLessons = $course->modules->flatMap->lessons->values();
		$totalLessons = $allLessons->count();
		$progressPercent = (int) ($enrollment->progress_percent ?? 0);
		$doneCount = $totalLessons ? (int) floor(($progressPercent / 100) * $totalLessons) : 0;

		$lessonStatuses = [];
		foreach ($allLessons as $idx => $lesson) {
			if ($idx < $doneCount) $lessonStatuses[$lesson->id] = 'done';
			elseif ($idx === $doneCount) $lessonStatuses[$lesson->id] = 'in_progress';
			else $lessonStatuses[$lesson->id] = 'locked';
		}

		$modulos = $course->modules->map(function($mod) use ($lessonStatuses){
			// Materiais por módulo: agregamos alguns dos materiais das lições do módulo
			$lessonIds = $mod->lessons->pluck('id');
			$materiaisModulo = LessonMaterial::whereIn('lesson_id', $lessonIds)->latest('updated_at')->take(6)->get()->map(function($m){
				$tipoMap = ['pdf' => 'PDF','link' => 'Link','doc' => 'Doc','code' => 'Código','guide' => 'Guia'];
				return [
					'tipo' => $tipoMap[strtolower($m->type ?? '')] ?? ucfirst($m->type ?? 'Arquivo'),
					'titulo' => $m->name,
					'tamanho' => $m->size_kb ? round($m->size_kb/1024,2).' MB' : null,
					'url' => $m->url,
				];
			})->all();

			return [
				'id' => (string) $mod->id,
				'titulo' => $mod->title,
				'licoes' => $mod->lessons->map(function($l) use ($lessonStatuses){
					return [
						'id' => (string) $l->id,
						'titulo' => $l->title,
						'duracao' => (int) ($l->duration_minutes ?? 0),
						'status' => $lessonStatuses[$l->id] ?? 'locked',
					];
				})->all(),
				'materiais' => $materiaisModulo,
			];
		})->all();

		$nextLesson = $allLessons->first(function($l) use ($lessonStatuses){ return ($lessonStatuses[$l->id] ?? null) !== 'done'; }) ?? $allLessons->last();
		$proxima = [
			'id' => $nextLesson?->id,
			'titulo' => $nextLesson?->title,
			'thumb' => $course->cover_url ?: 'https://images.unsplash.com/photo-1517433456452-f9633a875f6f?q=80&w=1200&auto=format&fit=crop',
			'duracao' => (int) ($nextLesson?->duration_minutes ?? 0),
		];

		$curso = [
			'id' => $course->id,
			'titulo' => $course->title,
			'capa' => $course->cover_url ?: 'https://images.unsplash.com/photo-1517433456452-f9633a875f6f?q=80&w=1200&auto=format&fit=crop',
		];

		return view('aluno.curso-andamento', compact('curso','modulos','proxima'));
	}

	private function findCourse(string $id): ?Course
	{
		if (ctype_digit($id)) return Course::with('modules.lessons')->find($id);
		return Course::with('modules.lessons')->where('slug', $id)->first();
	}
}
