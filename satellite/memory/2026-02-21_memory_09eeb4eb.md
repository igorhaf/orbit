<?php

namespace App\Http\Controllers\Aluno;

use App\Http\Controllers\Controller;
use App\Models\Enrollment;
use Illuminate\Http\Request;

class FaturasController extends Controller
{
	public function index(Request $request)
	{
		$user = $request->user();
		$faturas = Enrollment::with('course')
			->where('user_id', $user->id)
			->orderByDesc('created_at')
			->get()
			->map(function(Enrollment $e){
				return [
					'ref' => $e->ref ?? ('FT-'.$e->id),
					'curso' => $e->course?->title,
					'cursoId' => $e->course?->id,
					'data' => optional($e->created_at)?->toDateString(),
					'status' => $e->status ?? 'paga',
					'valor' => ($e->subtotal_cents ?? 0) / 100,
				];
			})->all();

		$anoAtual = now()->year;
		$totalPagoAno = collect($faturas)->filter(fn($f)=> (int)substr($f['data']??'',0,4) === $anoAtual && $f['status']==='paga')
			->sum(fn($f)=> $f['valor']);
		$proxima = collect($faturas)->firstWhere('status','pendente');
		$metodos = [];

		return view('aluno.faturas', compact('faturas','anoAtual','totalPagoAno','proxima','metodos'));
	}

	public function show(Request $request, string $ref)
	{
		$user = $request->user();
		$e = Enrollment::with('course')
			->where('user_id', $user->id)
			->where(function($q) use ($ref){ $q->where('ref',$ref)->orWhere('id', (int)str_replace('FT-','',$ref)); })
			->firstOrFail();

		$fatura = [
			'ref' => $e->ref ?? ('FT-'.$e->id),
			'curso' => $e->course?->title,
			'cursoId' => $e->course?->id,
			'data' => optional($e->created_at)?->toDateString(),
			'status' => $e->status ?? 'paga',
			'valor' => ($e->subtotal_cents ?? 0) / 100,
			'desconto' => ($e->discount_cents ?? 0) / 100,
			'cupom' => $e->discount_cents ? 'CUPOM' : null,
			'total' => ($e->total_cents ?? $e->subtotal_cents ?? 0) / 100,
		];

		return view('aluno.fatura-detalhe', compact('fatura'));
	}
}
