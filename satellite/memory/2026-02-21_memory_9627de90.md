<?php

namespace App\Http\Controllers\Cursos;

use App\Http\Controllers\Controller;
use App\Models\Course;
use App\Models\LessonMaterial;
use Illuminate\Http\Request;

class MaterialsController extends Controller
{
	public function index(Request $request, string $id)
	{
		$course = $this->findCourse($id);
		abort_unless($course, 404);
		$curso = [ 'id' => $course->id, 'titulo' => $course->title ];

		$items = LessonMaterial::query()
			->whereIn('lesson_id', $course->modules()->with('lessons:id,module_id')->get()->flatMap->lessons->pluck('id'))
			->latest('updated_at')
			->take(100)
			->get()
			->map(function(LessonMaterial $m){
				$tipoMap = [ 'pdf' => 'PDFs', 'link' => 'Links', 'doc' => 'Docs', 'code' => 'Código', 'guide' => 'Guias' ];
				$tipo = $tipoMap[strtolower($m->type ?? '')] ?? ucfirst($m->type ?? 'Arquivo');
				return [
					'id' => $m->id,
					'tipo' => $tipo,
					'titulo' => $m->name,
					'tag' => $m->tags[0] ?? null,
					'novo' => false,
					'recomendado' => false,
					'tamanho' => $m->size_kb ? round($m->size_kb/1024,2).' MB' : '—',
					'url' => $m->url,
				];
			})->all();

		return view('cursos.materiais', compact('curso','items'));
	}

	private function findCourse(string $id): ?Course
	{
		if (ctype_digit($id)) return Course::find($id);
		return Course::where('slug', $id)->first();
	}
}
