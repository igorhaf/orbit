<?php

namespace App\Http\Controllers;

use App\Models\Plan;
use Illuminate\Http\Request;

class PlanosController extends Controller
{
    public function index(Request $request)
    {
        $periodo = $request->query('periodo', 'mensal'); // mensal|anual

        $planos = Plan::orderBy('sort_order')
            ->get()
            ->map(function (Plan $plan) use ($periodo) {
                $precoCents = $periodo === 'anual' ? $plan->annual_price_cents : $plan->monthly_price_cents;
                return [
                    'id' => $plan->id,
                    'slug' => $plan->slug,
                    'nome' => $plan->name,
                    'subtitulo' => $plan->subtitle,
                    'preco' => $precoCents / 100,
                    'periodo' => $periodo,
                    'features' => $plan->features ?? [],
                    'highlight' => (bool) $plan->highlight,
                    'recommended' => (bool) $plan->recommended,
                    'badge' => $plan->badge,
                    'trial_days' => $plan->trial_days,
                ];
            })
            ->all();

        return view('planos.index', compact('planos', 'periodo'));
    }
}
