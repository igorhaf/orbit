<?php

namespace App\Http\Controllers\Instrutor;

use App\Http\Controllers\Controller;
use App\Models\Course;
use App\Models\Module;
use App\Models\Lesson;
use Illuminate\Http\Request;

class EstruturaController extends Controller
{
    public function show(Request $request, string $id)
    {
        $user = $request->user();
        if (!$user || $user->role !== 'instrutor') {
            abort(403);
        }

        $course = Course::where('id', $id)
            ->where('instructor_id', $user->id)
            ->with(['modules.lessons.materials'])
            ->firstOrFail();

        $curso = [
            'id' => $course->id,
            'titulo' => $course->title,
            'cover' => $course->cover_url,
            'status' => $course->status,
        ];

        $modulos = $course->modules->map(function (Module $module) {
            return [
                'id' => $module->id,
                'titulo' => $module->title,
                'descricao' => $module->description,
                'ordem' => $module->order,
                'licoes' => $module->lessons->map(function (Lesson $lesson) {
                    return [
                        'id' => $lesson->id,
                        'titulo' => $lesson->title,
                        'tipo' => $lesson->type ?? 'video',
                        'duracao' => $lesson->duration_minutes,
                        'status' => $lesson->status ?? 'rascunho',
                        'tem_material' => $lesson->materials->isNotEmpty(),
                        'ordem' => $lesson->order,
                    ];
                })->sortBy('order')->values()->all(),
            ];
        })->sortBy('order')->values()->all();

        $progresso = $this->calcularProgresso($course);

        return view('instrutor.cursos.estrutura', compact('curso', 'modulos', 'progresso'));
    }

    private function calcularProgresso(Course $course): array
    {
        $totalLicoes = $course->modules->sum(function ($module) {
            return $module->lessons->count();
        });

        $licoesComMaterial = $course->modules->sum(function ($module) {
            return $module->lessons->sum(function ($lesson) {
                return $lesson->materials->isNotEmpty() ? 1 : 0;
            });
        });

        $licoesPublicadas = $course->modules->sum(function ($module) {
            return $module->lessons->where('status', 'publicado')->count();
        });

        return [
            'total_licoes' => $totalLicoes,
            'com_material' => $licoesComMaterial,
            'publicadas' => $licoesPublicadas,
            'percentual_material' => $totalLicoes > 0 ? round(($licoesComMaterial / $totalLicoes) * 100) : 0,
            'percentual_publicadas' => $totalLicoes > 0 ? round(($licoesPublicadas / $totalLicoes) * 100) : 0,
        ];
    }
}
