<?php

namespace App\Http\Controllers\Instrutor;

use App\Http\Controllers\Controller;
use App\Models\Course;
use App\Models\LessonDiscussion;
use Illuminate\Http\Request;

class ComunidadeController extends Controller
{
    public function show(Request $request, string $id)
    {
        $user = $request->user();
        if (!$user || $user->role !== 'instrutor') {
            abort(403);
        }

        $course = Course::where('id', $id)
            ->where('instructor_id', $user->id)
            ->firstOrFail();

        $curso = [
            'id' => $course->id,
            'titulo' => $course->title,
            'cover' => $course->cover_url,
        ];

        // Buscar discussões das lições do curso
        $lessonIds = $course->modules->flatMap(function ($module) {
            return $module->lessons->pluck('id');
        });

        $discussoes = LessonDiscussion::whereIn('lesson_id', $lessonIds)
            ->with(['lesson.module', 'user'])
            ->latest()
            ->get()
            ->map(function (LessonDiscussion $discussion) {
                return [
                    'id' => $discussion->id,
                    'titulo' => $discussion->title,
                    'descricao' => $discussion->body,
                    'tags' => $discussion->tags ?? [],
                    'votos' => $discussion->votes_count ?? 0,
                    'respostas' => [], // Por enquanto vazio, pode ser expandido depois
                    'autor' => $discussion->user?->name ?? 'Usuário',
                    'tempo' => $discussion->created_at->diffForHumans(),
                    'resolvido' => !is_null($discussion->resolved_at),
                    'licao' => [
                        'titulo' => $discussion->lesson->title,
                        'modulo' => $discussion->lesson->module->title,
                    ],
                ];
            })
            ->all();

        $todasTags = collect($discussoes)->flatMap(fn($t) => $t['tags'])->unique()->values()->all();

        $resumo = [
            'total' => count($discussoes),
            'resolvidas' => collect($discussoes)->where('resolvido', true)->count(),
            'pendentes' => collect($discussoes)->where('resolvido', false)->count(),
            'tags_populares' => collect($discussoes)->flatMap(fn($t) => $t['tags'])->countBy()->sortDesc()->take(5)->keys()->all(),
        ];

        return view('instrutor.cursos.comunidade', compact('curso', 'discussoes', 'todasTags', 'resumo'));
    }
}
