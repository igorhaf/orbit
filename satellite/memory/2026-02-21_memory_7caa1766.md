Interview Query Classifier - PROMPT #231

Zero-latency heuristic classifier for interview queries.
Determines question complexity, estimated output tokens, and recommended
model tier WITHOUT making any AI calls.

Used by the Router utility node to skip cheap models for complex queries
and by adaptive timeout to estimate response time.
"""

import re
import logging
from typing import Dict, List, Optional

logger = logging.getLogger(__name__)

# ---------------------------------------------------------------------------
# Technical keyword sets
# ---------------------------------------------------------------------------

TECHNICAL_KEYWORDS = {
    "architecture", "database", "api", "deploy", "deployment", "scale", "scaling",
    "security", "performance", "infrastructure", "microservice", "monolith",
    "kubernetes", "docker", "ci/cd", "pipeline", "monitoring", "logging",
    "caching", "load balancer", "sharding", "replication", "migration",
    "authentication", "authorization", "oauth", "jwt", "encryption",
    "websocket", "graphql", "rest", "grpc", "queue", "kafka", "rabbitmq",
    "redis", "elasticsearch", "terraform", "aws", "gcp", "azure",
    "serverless", "lambda", "cdn", "dns", "ssl", "tls", "firewall",
    "backup", "disaster recovery", "high availability", "failover",
    "algoritmo", "arquitetura", "banco de dados", "seguranca", "desempenho",
    "escalabilidade", "infraestrutura", "autenticação", "autorizacao",
    "implantacao", "monitoramento",
}

REASONING_PATTERNS = [
    re.compile(r"\b(?:why|por\s?que|como|how\s+would|explain|compare|trade.?off|pros?\s+(?:and|e)\s+cons?)\b", re.IGNORECASE),
    re.compile(r"\b(?:what\s+(?:are|is)\s+the\s+(?:best|ideal|recommended|optimal))\b", re.IGNORECASE),
    re.compile(r"\b(?:qual\s+(?:a\s+melhor|o\s+melhor|seria))\b", re.IGNORECASE),
    re.compile(r"\b(?:descreva|detalhe|explique|justifique|análise)\b", re.IGNORECASE),
    re.compile(r"\b(?:describe|detail|justify|analyze|evaluate|assess)\b", re.IGNORECASE),
]

# Token estimates per complexity level
TOKEN_ESTIMATES = {
    "simple": 200,
    "moderate": 500,
    "complex": 1000,
}

# Tier mapping
TIER_MAP = {
    "simple": "fast",
    "moderate": "balanced",
    "complex": "strong",
}


def classify_interview_query(
    question_text: str,
    options: Optional[List[str]] = None,
    conversation_history_length: int = 0,
) -> Dict:
    """
    Classify an interview query by complexity without using AI.

    Args:
        question_text: The question being asked
        options: Pre-defined answer options (if closed question)
        conversation_history_length: Number of previous messages in conversation

    Returns:
        {
            "complexity": "simple"|"moderate"|"complex",
            "estimated_output_tokens": int,
            "needs_reasoning": bool,
            "is_technical": bool,
            "recommended_tier": "fast"|"balanced"|"strong",
        }
    """
    text_lower = question_text.lower()
    has_options = options is not None and len(options) > 0

    # Check technical keywords
    is_technical = any(kw in text_lower for kw in TECHNICAL_KEYWORDS)

    # Check reasoning patterns
    needs_reasoning = any(pat.search(question_text) for pat in REASONING_PATTERNS)

    # If it's an open question (no options), it likely needs more reasoning
    if not has_options:
        needs_reasoning = True

    # Determine complexity
    if has_options and len(options) <= 5 and not is_technical:
        complexity = "simple"
    elif is_technical and needs_reasoning:
        complexity = "complex"
    elif is_technical or needs_reasoning:
        complexity = "moderate"
    else:
        complexity = "simple"

    # Adjust for deep conversations (later questions tend to be more complex)
    if conversation_history_length > 12 and complexity == "simple":
        complexity = "moderate"
    elif conversation_history_length > 20 and complexity == "moderate":
        complexity = "complex"

    estimated_tokens = TOKEN_ESTIMATES[complexity]
    recommended_tier = TIER_MAP[complexity]

    result = {
        "complexity": complexity,
        "estimated_output_tokens": estimated_tokens,
        "needs_reasoning": needs_reasoning,
        "is_technical": is_technical,
        "recommended_tier": recommended_tier,
    }

    logger.info(
        f"🧠 Query classifier: complexity={complexity}, tier={recommended_tier}, "
        f"technical={is_technical}, reasoning={needs_reasoning}, "
        f"est_tokens={estimated_tokens}"
    )

    return result
