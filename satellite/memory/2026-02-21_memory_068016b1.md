<?php

namespace App\Http\Controllers\Trilhas;

use App\Http\Controllers\Controller;
use App\Models\Course;
use Illuminate\Http\Request;
use Illuminate\Support\Str;

class ExplorarController extends Controller
{
	public function index(Request $request)
	{
		$courses = Course::query()->published()->get(['id','title','tags','duration_minutes','level','rating_average','cover_url','language']);
		$tagToCourses = [];
		foreach ($courses as $course) {
			foreach ((array)($course->tags ?? []) as $tag) {
				$k = trim($tag); if ($k==='') continue;
				$tagToCourses[$k] = $tagToCourses[$k] ?? [];
				$tagToCourses[$k][] = $course;
			}
		}
		$trilhas = [];
		foreach ($tagToCourses as $tag => $list) {
			$durMin = collect($list)->sum('duration_minutes');
			$durHoras = (int) round($durMin/60);
			$nivel = collect($list)->pluck('level')->filter()->countBy()->sortDesc()->keys()->first() ?? 'Intermediário';
			$nota = (float) round(collect($list)->avg('rating_average'), 1);
			$id = Str::slug($tag);
			$idioma = collect($list)->pluck('language')->filter()->countBy()->sortDesc()->keys()->first() ?? 'pt';
			$objetivo = in_array(strtolower($tag), ['carreira','profissional']) ? 'Carreira' : 'Habilidade';
			$capa = collect($list)->pluck('cover_url')->filter()->first() ?? 'https://picsum.photos/seed/'.(700 + (crc32($id)%300)).'/1200/675';
			$trilhas[] = [
				'id' => $id,
				'nome' => 'Trilha '.$tag,
				'resumo' => 'Cursos focados em '.$tag.'.',
				'cursos' => count($list),
				'duracao' => max(1,$durHoras),
				'nivel' => $nivel,
				'nota' => $nota,
				'idioma' => $idioma==='en'?'Inglês':($idioma==='es'?'Espanhol':'Português'),
				'objetivo' => $objetivo,
				'categoriaDuracao' => $durHoras<=10 ? 'Curta (<= 10h)' : ($durHoras<=30 ? 'Média (10–30h)' : 'Longa (30h+)'),
				'certificado' => true,
				'popularidade' => min(99, 40 + $durHoras*2),
				'criadoEm' => now()->subDays(mt_rand(10,200))->toDateString(),
				'recomendado' => $nota>=4.5 || $durHoras>=20,
				'capa' => $capa,
			];
		}

		$objetivos = ['Carreira','Habilidade'];
		$niveis = ['Iniciante','Intermediário','Avançado'];
		$duracoes = ['Curta (<= 10h)','Média (10–30h)','Longa (30h+)'];
		$idiomas = ['Português','Inglês','Espanhol'];

		return view('trilhas.explorar', compact('trilhas','objetivos','niveis','duracoes','idiomas'));
	}
}
