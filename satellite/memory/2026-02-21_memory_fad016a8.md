<?php

namespace App\Http\Controllers\Categorias;

use App\Http\Controllers\Controller;
use App\Models\Course;
use Illuminate\Http\Request;
use Illuminate\Support\Str;

class IndexController extends Controller
{
	public function index(Request $request)
	{
		$courses = Course::query()->published()->select('id','title','category','tags','cover_url')->get();

		$tagToCourses = [];
		foreach ($courses as $course) {
			$tags = is_array($course->tags) ? $course->tags : [];
			foreach ($tags as $tag) {
				$k = trim((string) $tag);
				if ($k==='') continue;
				$tagToCourses[$k] = $tagToCourses[$k] ?? [];
				$tagToCourses[$k][] = $course;
			}
		}

		$categorias = [];
		$areasSet = [];
		foreach ($tagToCourses as $tag => $list) {
			$area = collect($list)->pluck('category')->filter()->countBy()->sortDesc()->keys()->first() ?? 'Geral';
			$areasSet[$area] = true;
			$img = collect($list)->pluck('cover_url')->filter()->first();
			if (!$img) {
				$seed = 300 + (crc32(Str::slug($tag)) % 700);
				$img = "https://picsum.photos/seed/{$seed}/1200/675";
			}
			$categorias[] = [
				'id' => Str::slug($tag),
				'nome' => $tag,
				'area' => $area,
				'qtd' => count($list),
				'img' => $img,
			];
		}

		usort($categorias, fn($a,$b)=> strcasecmp($a['nome'],$b['nome']));
		$areas = array_keys($areasSet);
		sort($areas, SORT_FLAG_CASE | SORT_NATURAL);

		return view('categorias.index', compact('areas','categorias'));
	}
}
