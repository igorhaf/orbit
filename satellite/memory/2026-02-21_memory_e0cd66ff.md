<?php

namespace App\Http\Controllers\Aluno;

use App\Http\Controllers\Controller;
use App\Models\Enrollment;
use Illuminate\Http\Request;

class InscricaoStatusController extends Controller
{
	public function show(Request $request, string $id)
	{
		$user = $request->user();
		$e = Enrollment::with('course')->where('user_id', $user->id)->whereHas('course', fn($q)=> $q->where('id', $id))->firstOrFail();
		$inscricao = [
			'id' => $id,
			'curso' => $e->course?->title,
			'capa' => $e->course?->cover_url,
			'passos' => ['Pedido criado','Processamento','Confirmado'],
			'estado' => $e->status === 'paid' ? 'confirmado' : ($e->status === 'cancelled' ? 'erro' : 'pendente'),
			'valores' => [
				'subtotal' => ($e->subtotal_cents ?? 0) / 100,
				'desconto' => ($e->discount_cents ?? 0) / 100,
				'cupom' => $e->discount_cents ? 'CUPOM' : null,
				'total' => ($e->total_cents ?? $e->subtotal_cents ?? 0) / 100,
			],
		];
		return view('aluno.inscricao-status', compact('inscricao','id'));
	}
}
